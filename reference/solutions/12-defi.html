练习 1：从 0.3% 手续费中抽取 1/6 给协议，并测试 k 与费用分配

在 MiniAMM 中增加协议费接收方与比例：

solidity
address public immutable feeRecipient;
uint256 public constant PROTOCOL_FEE_BPS = FEE_BPS / 6; // 5 bps = 0.05%

构造函数改为：

solidity
constructor(address token0_, address token1_, address feeRecipient_) {
    require(feeRecipient_ != address(0), 'zero fee recipient');
    token0 = IERC20(token0_);
    token1 = IERC20(token1_);
    feeRecipient = feeRecipient_;
}

swap 改为：

solidity
function swap(uint256 amountIn, bool zeroForOne, uint256 minAmountOut)
    external
    returns (uint256 amountOut)
{
    if (amountIn == 0) revert ZeroAmount();

    (uint256 reserveIn, uint256 reserveOut) =
        zeroForOne ? (reserve0, reserve1) : (reserve1, reserve0);
    if (reserveIn == 0 || reserveOut == 0) revert NoLiquidity();

    (IERC20 tokenIn, IERC20 tokenOut) = zeroForOne ? (token0, token1) : (token1, token0);

    tokenIn.transferFrom(msg.sender, address(this), amountIn);

    // 用户仍然按 0.3% 的总费率得到报价
    uint256 amountInWithFee = amountIn * (BPS - FEE_BPS);
    amountOut = (reserveOut * amountInWithFee) / (reserveIn * BPS + amountInWithFee);

    if (amountOut == 0) revert ZeroOutput();
    if (amountOut < minAmountOut) revert SlippageExceeded(minAmountOut, amountOut);

    // 协议从总手续费中拿走 1/6，剩余 5/6 留给 LP
    uint256 protocolFee = (amountIn * PROTOCOL_FEE_BPS) / BPS;

    tokenOut.transfer(msg.sender, amountOut);
    if (protocolFee > 0) tokenIn.transfer(feeRecipient, protocolFee);

    _updateReserves();
    emit Swap(msg.sender, zeroForOne, amountIn, amountOut);
}

要点：

  • 总费率仍是 0.3%，只是其中 0.05% 归协议、0.25% 归 LP；用户报价不变。
  • 协议费在输入代币上收取，并在 _updateReserves() 之前转出，因此储备不会把这笔钱算成 LP 的资产。
  • k 仍然单调不减，但增幅比“全部手续费归 LP”小。测试应断言“不减”，不要断言一个固定增幅。
  • 如果 feeRecipient 是拒收代币的合约，swap 会整体回滚。生产环境应使用可信收款方或 pull payment。
  • 费率变更应受治理与上限约束，并加事件；本示例把 5 bps 写死为教学值。
练习 2：给 MiniLending 增加按时间累计的利息

在 MiniLending 中增加：

solidity
uint256 public constant INTEREST_RATE_BPS_PER_DAY = 100; // 1%/天，教学参数
mapping(address => uint256) public lastAccrual;

event InterestAccrued(address indexed user, uint256 interest);

增加两个函数：

solidity
function currentDebt(address user) public view returns (uint256) {
    uint256 debt = debtBalance[user];
    if (debt == 0) return 0;

    uint256 elapsed = block.timestamp - lastAccrual[user];
    return debt + (debt * INTEREST_RATE_BPS_PER_DAY * elapsed) / (BPS * 1 days);
}

function _accrue(address user) private returns (uint256 debt) {
    uint256 previous = debtBalance[user];
    debt = currentDebt(user);

    if (debt > previous) {
        emit InterestAccrued(user, debt - previous);
        debtBalance[user] = debt;
    }
    lastAccrual[user] = block.timestamp;
}

function healthFactor(address user) public view returns (uint256) {
    uint256 debt = currentDebt(user);
    if (debt == 0) return type(uint256).max;

    return
        (collateralValue(user) * LIQUIDATION_THRESHOLD_BPS * PRICE_SCALE) /
        (BPS * debt);

在 borrow、repay、withdrawCollateral、liquidate 的开头调用 _accrue：

solidity
function borrow(uint256 amount) external {
    if (amount == 0) revert ZeroAmount();

    _accrue(msg.sender);

    uint256 limit = (collateralValue(msg.sender) * LTV_BPS) / BPS;
    uint256 requested = debtBalance[msg.sender] + amount;
    if (requested > limit) revert ExceedsLtv(limit, requested);

    debtBalance[msg.sender] = requested;
    lastAccrual[msg.sender] = block.timestamp;
    debtToken.transfer(msg.sender, amount);
    emit Borrowed(msg.sender, amount);
}

repay 与 liquidate 中把债务读取改成累计后的值，例如：

solidity
function repay(uint256 amount) external {
    if (amount == 0) revert ZeroAmount();
    _accrue(msg.sender);

    if (debtBalance[msg.sender] < amount) {
        revert InsufficientDebt(debtBalance[msg.sender], amount);
    }

    debtBalance[msg.sender] -= amount;
    debtToken.transferFrom(msg.sender, address(this), amount);
    emit Repaid(msg.sender, amount);
}

import { time } from '@nomicfoundation/hardhat-network-helpers'

it('利息累计会降低健康因子并触发清算', async () => {
  const { lending, alice, liquidator } = await deployLendingFixture()

要点：

  • 这是单利模型，只在交互时结算，不在每个区块复利；生产借贷通常用全局借款指数与 accrualIndex，避免每个用户循环。
  • healthFactor 必须读取“当前应付债务”，否则时间过去后前端仍显示健康，清算人会错过窗口。
  • withdrawCollateral 也要先 _accrue，否则用户可以在利息累积后按旧债务通过健康检查。
  • 升级已有合约时，lastAccrual 对新用户默认为 0 会导致巨额利息。真实迁移要先初始化所有存量用户的 lastAccrual，或引入全局指数。
  • 参数 1%/天只用于测试快速触发；生产利率应由利用率曲线与治理决定。
练习 3：模拟同一区块内 swap 后按现货价清算

这段测试放在 examples/12-defi，复用 MiniAMM 与 deployAmmFixture：

ts
it('一笔 swap 压低现货价，按该价格清算会产生坏账', async () => {
  const { amm, bob } = await deployAmmFixture()
  const spot = async () => ((await amm.reserve1()) * E18) / (await amm.reserve0())
  const fair = await spot()

  // bob 用 300 COL 砸盘
  await amm.connect(bob).swap(300n * E18, true, 0n)
  const manipulated = await spot()

  const collateral = 10n * E18
  const debt = 7n * E18
  const manipulatedHealth =
    (collateral * manipulated * 8_000n) / (10_000n * debt)

  expect(manipulated).to.be.lessThan((fair * 7n) / 10n)
  expect(manipulatedHealth).to.be.lessThan(E18)

  // 部分清算后，按真实价格计算已经资不抵债
  const repay = 5n * E18
  const seized = (repay * 10_500n * E18) / (10_000n * manipulated)
  const remainDebt = debt - repay
  const remainCollateral = collateral - seized
  const badDebt = remainDebt - (remainCollateral * fair) / E18

  expect(seized).to.be.lessThan(collateral)
  expect(badDebt).to.be.greaterThan(0n)
})

要点：

  • 攻击者不需要改预言机，只需要在借贷协议读取的同一个池子里做一笔大额 swap。
  • 如果清算交易和 swap 在同一个区块（甚至同一个交易）执行，协议看到的是被操纵的瞬时价格。
  • 部分清算后，借款人剩余抵押按真实价格已经不够还剩余债务，差额就是坏账。
  • 真实攻击常配合闪电贷放大资金量，进一步压低价格。
  • 防守：TWAP、多源中位数、价格偏离阈值、单区块价格变化上限、清算延迟、坏账准备金、暂停开关。
  • 测试里用公式模拟 lending；接入真实合约时保持同一个价格源与同一组断言即可。

要把“同一交易”真正落地，可以再加一个执行器合约，把 swap 和清算放进同一个函数：

solidity
interface ILiquidator {
    function liquidate(address user, uint256 repayAmount) external;
}

contract SpotAttackExecutor {
    MiniAMM public immutable amm;
    ILiquidator public immutable lending;

    constructor(MiniAMM amm_, ILiquidator lending_) {
        amm = amm_;
        lending = lending_;
    }

    function attack(address user, uint256 amountIn, uint256 repayAmount) external {
        // 同一个交易内：先操纵价格，再触发清算
        amm.swap(amountIn, true, 0);
        lending.liquidate(user, repayAmount);
    }
}

练习 4：抵押记账不变量 sum(collateralBalance) ≤ 实际余额

在 MiniLending 中增加总量：

uint256 public totalCollateral;

    return totalCollateral <= collateral.balanceOf(address(this));

要点：

  • 正常路径下是等式：totalCollateral == collateral.balanceOf(lending)。
  • 直接给合约转抵押代币会让实际余额大于记账，用 <= 仍然安全，但会暴露“有人捐赠”的情况。
  • 如果支持 fee-on-transfer 或 rebasing 代币，实际到账可能小于 amount，等式会被打破，甚至出现记账大于余额。此时必须按实际到账量记账，或者直接不支持这类代币。
  • 迁移已有合约时，totalCollateral 需要从 0 开始并按存量用户补账，否则不变量一开始就是错的。
  • 这只是单项不变量；完整借贷还应检查债务总额、价格源、清算后抵押与债务的对应关系。
