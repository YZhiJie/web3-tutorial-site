练习 1：给 MiniToken 增加 burn(uint256)，只能销毁自己的余额

在合约中增加：

solidity
function burn(uint256 amount) external {
    uint256 balance = balanceOf[msg.sender];
    if (balance < amount) revert InsufficientBalance(msg.sender, balance, amount);

    balanceOf[msg.sender] = balance - amount;
    totalSupply -= amount;

    emit Transfer(msg.sender, address(0), amount);
}

练习 2：零地址 revert，转给自己直接返回 true

把 transfer 改成：

solidity
function transfer(address to, uint256 amount) external returns (bool) {
    if (to == address(0)) revert ZeroAddress();
    if (to == msg.sender) return true;

    _transfer(msg.sender, to, amount);
    return true;
}

  expect(

  ).to.equal(true)

  await token.connect(alice).transfer(alice.address, 50n)
  expect(await token.balanceOf(alice.address)).to.equal(100n)

练习 3：实现 batchTransfer，并说明为什么必须校验数组长度

合约增加：

solidity
error LengthMismatch(uint256 toLength, uint256 amountsLength);

function batchTransfer(
    address[] calldata to,
    uint256[] calldata amounts
) external {
    if (to.length != amounts.length) {
        revert LengthMismatch(to.length, amounts.length);
    }

    for (uint256 i = 0; i < to.length; i++) {
        if (to[i] == address(0)) revert ZeroAddress();
        _transfer(msg.sender, to[i], amounts[i]);
    }
}

it('batchTransfer 拒绝长度不一致的数组', async () => {

    .to.be.revertedWithCustomError(token, 'LengthMismatch')
    .withArgs(1n, 2n)

it('batchTransfer 依次转账', async () => {
  const { token, alice, bob } = await deploy()

  await token.batchTransfer([alice.address, bob.address], [10n, 20n])

  expect(await token.balanceOf(alice.address)).to.equal(10n)
  expect(await token.balanceOf(bob.address)).to.equal(20n)

为什么必须校验长度：

  • 如果 amounts 比 to 短，循环访问 amounts[i] 会越界，直接 revert，错误信息不直观。
  • 如果 amounts 比 to 长，多出来的金额会被静默忽略，用户以为转了 N 笔，实际只转了前几笔。
  • 显式校验能在入口给出确定的错误，并且整笔交易原子回滚，不会出现“前几笔成功、后面失败”的中间状态。

批量转账仍然是一笔交易，任何一项失败都会让整批回滚；但 Gas 已经消耗，不要把它当成“失败免费”。

topics[0] 是事件签名的 keccak256，不包含参数名；topics[1..3] 是 indexed 参数，最多三个；data 按 ABI 顺序存放非 indexed 参数。前端解析时，参数顺序必须与 ABI 完全一致。
