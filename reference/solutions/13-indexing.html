练习 1：给 LogIndexer 增加失败重试与指数退避

修改 src/indexer.mjs 的构造函数与 scan：

js
export class LogIndexer {
  constructor({
    getLogs,
    fromBlock = 0,
    pageSize = 1000,
    retries = 3,
    baseDelayMs = 100,
    sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms))
  }) {
    if (typeof getLogs !== 'function') throw new Error('getLogs is required')
    if (pageSize <= 0) throw new Error('pageSize must be positive')
    if (retries < 0) throw new Error('retries must be >= 0')

    this.getLogs = getLogs
    this.fromBlock = fromBlock
    this.pageSize = pageSize
    this.retries = retries
    this.baseDelayMs = baseDelayMs
    this.sleep = sleep
  }

  async #fetchPage(fromBlock, toBlock) {
    let lastError

    for (let attempt = 0; attempt <= this.retries; attempt += 1) {
      try {
        return await this.getLogs({ fromBlock, toBlock })
      } catch (error) {
        lastError = error
        if (attempt === this.retries) break
        await this.sleep(this.baseDelayMs * 2 ** attempt)
      }
    }

    throw lastError
  }

  async scan(toBlock) {
    if (toBlock < this.fromBlock) {
      return { logs: [], ranges: [], nextFromBlock: this.fromBlock }
    }

    const logs = []
    const ranges = []

    for (let start = this.fromBlock; start <= toBlock; start += this.pageSize) {
      const end = Math.min(start + this.pageSize - 1, toBlock)
      const page = await this.#fetchPage(start, end)
      ranges.push([start, end])
      logs.push(...page)
    }

    this.fromBlock = toBlock + 1

    return { logs: normalize(logs), ranges, nextFromBlock: this.fromBlock }
  }
}

测试：

js
test('单页失败两次后重试成功，并按指数退避', async () => {
  let calls = 0
  const getLogs = async () => {
    calls += 1
    if (calls <= 2) throw new Error('rpc timeout')
    return [transferLog({ block: 1, index: 0 })]
  }

  const sleeps = []
  const indexer = new LogIndexer({
    getLogs,
    fromBlock: 0,
    pageSize: 10,
    retries: 2,
    baseDelayMs: 5,
    sleep: async (ms) => {
      sleeps.push(ms)
    }
  })

  const { logs } = await indexer.scan(9)

  assert.equal(calls, 3)
  assert.deepEqual(sleeps, [5, 10])
  assert.equal(logs.length, 1)
})

要点：

  • retries: 2 表示最多 3 次尝试；sleeps 依次是 5ms、10ms。
  • 注入 sleep 让测试不真正等待，也不会让测试变慢。
  • 只有整页成功后才推进 this.fromBlock；重试用尽时 scan 抛出，游标不前进，下一次从原位置重扫。
  • 重扫可能产生重复日志，normalize 用 txHash + logIndex 去重，因此重试是幂等的。
  • 生产中还应加超时、抖动（jitter）、最大退避上限，并区分“可重试的网络错误”和“不可重试的参数错误”。
练习 2：实现 transferHistory(address)，按 from/to 过滤

在 src/decode.mjs 增加：

js
export function transferHistory(logs, address) {
  const target = address.toLowerCase()

  return logs
    .map(decodeTransferLog)
    .filter(
      (transfer) =>
        transfer && (transfer.from === target || transfer.to === target)
    )
    .sort(
      (a, b) =>
        a.blockNumber - b.blockNumber || a.logIndex - b.logIndex
    )
}

要点：

  • decodeTransferLog 已经把地址转成小写，过滤前也要把查询地址转小写，否则大小写混用会漏数据。
  • 入账和出账都算“与地址相关”；如果只需要入账，改成只匹配 to。
  • 排序必须带 logIndex 作为第二关键字，同一区块内多笔转账才有稳定顺序。
  • 如果日志量很大，应把过滤下推到数据库查询，而不是在内存里全量 map。
  • 该函数返回的是解码后的对象，不是原始 log；需要原始数据时保留 transactionHash 与 logIndex 以便回溯。
练习 3：设计检查点格式与崩溃恢复流程

推荐的检查点 JSON：

json
{
  "chainId": 11155111,
  "contract": "0x1234...abcd",
  "topic0": "0xddf252ad1be2c89b69c2b068fc378daa952ba7f163c4a11628f55a4df523b3ef",
  "lastScannedBlock": 8123456,
  "lastFinalizedBlock": 8123432,

  "updatedAt": "2026-11-04T00:00:00.000Z"

字段含义：

  • lastScannedBlock：最后一个“日志已经落库”的区块。重启后从 lastScannedBlock + 1 开始。
  • lastFinalizedBlock：达到确认数、可以对外标记为 finalized 的区块。
  • reorgSafeBlock：允许发生重组的最早区块；发生 reorg 时从这里之后重扫。
  • chainId、contract、topic0：防止把不同链、不同合约、不同事件的检查点混在一起。

崩溃恢复规则：

  1. 每一页的日志和检查点必须在同一个数据库事务里提交。
  2. 事务提交前崩溃，检查点保持旧值，重启后重扫这一页；用 (txHash, logIndex) 唯一键去重。
  3. 事务提交后崩溃，日志与检查点一致，重启从下一页继续。
  4. 不要先更新检查点、再异步写日志；这会造成“游标跳过了但数据没落库”。
  5. 发生 reorg 时，删除 blockNumber >= reorgFromBlock 的旧日志，再从 min(reorgSafeBlock, reorgFromBlock - 1) + 1 重扫。
  6. 对外查询区分 pending 与 confirmed，不要把未达确认数的数据当成最终结果。

一个事务化写入的示意：

js
async function savePage(db, logs, checkpoint) {
  await db.transaction(async (tx) => {
    await tx.logs.insertMany(logs, { ignoreDuplicates: true })
    await tx.checkpoints.upsert(checkpoint)
  })
}

要点：

  • 检查点不是“越新越好”，而是“和已落库数据严格对应”。
  • 多实例部署时要用行锁或单写者，避免两个进程同时推进同一检查点。
  • 记录 updatedAt 与最后错误信息，方便告警和排查停滞。
  • 重组深度超过 reorgSafeBlock 时，需要人工介入或从更早的归档块重扫。
练习 4：测试未确认视图与已确认视图的边界
js
test('未确认与已确认视图在边界区块上不同', () => {
  const logs = [95, 96, 97, 98, 99, 100].map((block) =>
    transferLog({ block, index: 0 })
  )

  const pending = confirmedLogs(logs, 100, 0)
  const confirmed = confirmedLogs(logs, 100, 3)

  assert.deepEqual(
    pending.map((log) => log.blockNumber),
    [95, 96, 97, 98, 99, 100]
  )
  assert.deepEqual(
    confirmed.map((log) => log.blockNumber),
    [95, 96, 97]
  )
})

边界含义：

  • head = 100、confirmations = 3 时，安全区块是 100 - 3 = 97。
  • confirmedLogs 使用 <= safeBlock，所以 97 算已确认，98 不算。
  • confirmations = 0 时，当前 head 也包含在内，等价于“未确认视图”。
  • 前端应把 98–100 标成 pending，并允许它们随后被 reorg 替换。
  • 如果确认数改成 12，边界就是 88；测试要把 88 和 89 分别断言，避免差一错误。
