练习 1：给 MiniSafe 增加 owner 轮换，并要求多签确认

先去掉 threshold 的 immutable，增加 ownerCount 与“只能由多签自身调用”的修饰器：

solidity
uint256 public threshold;
uint256 public ownerCount;

error NotSelf();
error AlreadyOwner();
error ThresholdOutOfRange();

modifier onlySelf() {
    if (msg.sender != address(this)) revert NotSelf();
    _;
}

构造函数里补 ownerCount = 3，并保留原有阈值校验：

solidity
constructor(address[3] memory initialOwners, uint256 threshold_) {
    if (threshold_ == 0 || threshold_ > 3) revert NotEnoughConfirmations();
    for (uint256 i = 0; i < 3; i++) {
        if (initialOwners[i] == address(0)) revert ZeroAddress();
        owners[initialOwners[i]] = true;
    }
    threshold = threshold_;
    ownerCount = 3;
}

增加三个“只能被自身调用”的管理函数。外部 EOA 想改 owner，必须先走 propose → approve → execute：

solidity
function addOwner(address newOwner) external onlySelf {
    if (newOwner == address(0)) revert ZeroAddress();
    if (owners[newOwner]) revert AlreadyOwner();

    owners[newOwner] = true;
    ownerCount += 1;
}

function removeOwner(address owner) external onlySelf {
    if (!owners[owner]) revert NotOwner();
    if (ownerCount - 1 < threshold) revert NotEnoughConfirmations();

    owners[owner] = false;
    ownerCount -= 1;
}

function setThreshold(uint256 newThreshold) external onlySelf {
    if (newThreshold == 0 || newThreshold > ownerCount) revert ThresholdOutOfRange();
    threshold = newThreshold;
}

练习 2：给 timelock 增加最小延迟与最大延迟双边界

把构造参数从单个 delay 改成 minDelay 与 maxDelay：

solidity
uint256 public immutable minDelay;
uint256 public immutable maxDelay;

error DelayOutOfBounds();

constructor(address admin_, uint256 minDelay_, uint256 maxDelay_) {
    if (minDelay_ == 0 || maxDelay_ < minDelay_) revert DelayOutOfBounds();
    admin = admin_;
    minDelay = minDelay_;
    maxDelay = maxDelay_;
}

queue 同时检查上下界：

solidity
function queue(
    address target,
    uint256 value,
    bytes calldata data,
    uint256 eta
) external onlyAdmin returns (bytes32 id) {
    if (eta < block.timestamp + minDelay || eta > block.timestamp + maxDelay) {
        revert DelayOutOfBounds();
    }

    id = transactionId(target, value, data, eta);
    queued[id] = true;
    etaOf[id] = eta;

    emit Queued(id, target, value, eta);
}

练习 3：设计 guardian 多签（2-of-3），避免单点暂停

不需要新合约：把现成的 MiniSafe 当作 guardian 多签。金库的 guardian 指向这个多签合约，单个 guardian 无法直接暂停，必须凑齐 2 个确认。

ts
it('guardian 2-of-3 多签才能暂停金库', async () => {
  const { safe, deployer } = await loadFixture(deployFixture)
  const [g1, g2, g3] = (await ethers.getSigners()).slice(6, 9)

  const guardianSafe = await ethers.deployContract('MiniSafe', [
    [g1.address, g2.address, g3.address],
    2
  ])
  const treasury = await ethers.deployContract(
    'GovernedTreasury',
    [await safe.getAddress(), await guardianSafe.getAddress()],
    { value: ethers.parseEther('10') }
  )
  const pauseData = treasury.interface.encodeFunctionData('pause')

  await guardianSafe.connect(g1).propose(await treasury.getAddress(), 0n, pauseData)
  await guardianSafe.connect(g1).approve(0)

  await expect(
    guardianSafe.connect(g1).execute(0)
  ).to.be.revertedWithCustomError(guardianSafe, 'NotEnoughConfirmations')

  await guardianSafe.connect(g2).approve(0)
  await guardianSafe.connect(g1).execute(0)

  expect(await treasury.paused()).to.equal(true)
})

要点：

  • 3 个 guardian 使用独立密钥与独立设备，不要放在同一台机器。
  • guardian 多签只应拥有 pause 权限，不能转账，也不能解除暂停。
  • 解除暂停仍然走 owner（主多签 / timelock），保持非对称权限。
  • 生产环境把 guardian 合约升级、成员轮换写进 runbook，并做演练。
练习 4：写一个事件索引器，列出待执行的 timelock 提案

新建 examples/19-enterprise/scripts/pending-timelock.ts（本练习新增）。如果已经完成练习 2，把构造参数改成 [admin.address, 3000, 7200]：

ts
import { ethers } from 'hardhat'
import { time } from '@nomicfoundation/hardhat-toolbox/network-helpers'

async function main() {
  const [admin, recipient] = await ethers.getSigners()
  const timelock = await ethers.deployContract('Timelock', [admin.address, 3600])
  const deploymentBlock = (await timelock.deploymentTransaction())!.blockNumber

  const now = await time.latest()
  const eta1 = now + 4000
  const eta2 = eta1 + 100

  await timelock.connect(admin).queue(recipient.address, 1n, '0x', eta1)
  await timelock.connect(admin).queue(recipient.address, 2n, '0x', eta2)
  await timelock.connect(admin).cancel(recipient.address, 2n, '0x', eta2)

  const queuedEvents = await timelock.queryFilter(
    timelock.filters.Queued(),
    deploymentBlock
  )

  let pending = 0
  for (const event of queuedEvents) {
    const id = (event.args as any).id as string
    if (!(await timelock.queued(id))) continue

    pending += 1
    const eta = await timelock.etaOf(id)
    console.log(`pending id=${id} eta=${eta}`)
  }

  console.log(`待执行数量: ${pending}`)
}

main().catch((error) => {
  console.error(error)
  process.exitCode = 1
})

要“实时”监控，把一次性查询换成按块轮询；真实环境应记录上次处理的区块，避免重复与漏块：

ts
let lastBlock = await ethers.provider.getBlockNumber()

ethers.provider.on('block', async (blockNumber) => {
  if (blockNumber <= lastBlock) return

  const events = await timelock.queryFilter(
    timelock.filters.Queued(),
    lastBlock + 1,
    blockNumber
  )
  lastBlock = blockNumber

  for (const event of events) {
    const id = (event.args as any).id as string
    if (await timelock.queued(id)) {
      const eta = await timelock.etaOf(id)
      console.log(`pending id=${id} eta=${eta}`)
    }
  }
})

练习 5：写一份升级 runbook（提案、审查、执行、回滚、通知）

下面是一页可直接放进仓库的 runbook 模板。

角色与前置 ​

  • 提案人： 多签 owner 之一，负责写提案与提供变更说明。
  • 审查人： 至少 2 名未参与提案的工程师，检查字节码、存储布局、权限变更与测试。
  • 执行人： timelock 到期后由值班人员触发，任何人可验证。
  • 应急负责人： 持有 guardian 权限，只能在异常时暂停。

1. 提案 ​

  1. 在本地构建新实现合约，记录 commit、编译器版本、optimizer 设置与字节码哈希。
  2. 写出变更摘要：改了什么、为什么、影响哪些用户与资金。
  3. 附上升级兼容检查：存储布局、初始化函数、代理 admin、实现地址。
  4. 多签创建提案，提案内容在链下先做可读化审查，禁止盲签。

2. 审查 ​

  1. 对比新旧实现字节码与 ABI，确认没有意外函数。
  2. 跑完整测试、回归测试与升级后冒烟测试。
  3. 检查 timelock 队列里是否有其他待执行提案，避免组合冲突。
  4. 由未参与提案的审查人签字；高风险变更要求外部审计。

3. 执行 ​

  1. 确认 timelock eta 已到，且当前区块时间留有余量。
  2. 执行升级交易，保存交易哈希与区块号。
  3. 立即验证：实现地址、owner/admin、关键只读函数、暂停状态。
  4. 观察事件 Upgraded / AdminChanged 与监控告警。

4. 回滚 ​

  1. 预先准备旧实现地址与回滚提案，不等到出事才写。
  2. 回滚同样走 timelock；紧急情况可缩短延迟，但不能绕过暂停与多签。
  3. 回滚后重跑冒烟测试与资金守恒检查。
  4. 记录事故时间线、根因、影响范围与后续修复。

5. 通知 ​

  • 升级前：给用户与集成方发变更公告，写明时间窗与影响。
  • 执行后：发出交易哈希、区块号与验证结果。
  • 异常时：先暂停，再通知；不要让 guardian 直接转走资金。
  • 所有通知模板与联系人写进 runbook，不依赖个人记忆。

演练清单 ​

  • 私钥丢失时如何恢复多签？
  • guardian 被攻破时如何止损？
  • timelock 被塞入恶意提案时谁来看、多久能取消？
  • 升级后关键函数失败时如何回滚？

演练没有跑通，就不算有 runbook。
