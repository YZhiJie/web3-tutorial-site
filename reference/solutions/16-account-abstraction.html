练习 1：给 session key 增加函数选择器白名单

在 SmartAccount.sol 中给 SessionKey 增加 allowedSelector，并在验证时同时检查目标和选择器。

结构体与添加函数：

solidity
struct SessionKey {
    address allowedTarget;
    bytes4 allowedSelector;
    uint256 expiresAt;
}

function addSessionKey(
    address key,
    address allowedTarget,
    bytes4 allowedSelector,
    uint256 expiresAt
) external onlyOwner {
    if (
        key == address(0) ||
        allowedTarget == address(0) ||
        allowedSelector == bytes4(0) ||
        expiresAt <= block.timestamp
    ) {
        revert InvalidSessionKey();
    }
    sessionKeys[key] = SessionKey(allowedTarget, allowedSelector, expiresAt);
    emit SessionKeyAdded(key, allowedTarget, expiresAt);
}

it('session key 只能调用白名单函数选择器', async () => {
  const { entryPoint, account, counter, owner } = await loadFixture(deployFixture)
  const sessionKey = ethers.Wallet.createRandom()
  const expiresAt = (await time.latest()) + 3600
  const incrementSelector = counter.interface.getFunction('increment')!.selector

  await account
    .connect(owner)
    .addSessionKey(sessionKey.address, await counter.getAddress(), incrementSelector, expiresAt)

  const incrementData = counter.interface.encodeFunctionData('increment')
  const op = await buildOperation(await account.getAddress(), await counter.getAddress(), incrementData)
  await entryPoint.validateAndExecute(op, await signOperation(entryPoint, op, sessionKey))
  expect(await counter.value()).to.equal(1n)

  const resetData = counter.interface.encodeFunctionData('reset')
  const badOp = await buildOperation(await account.getAddress(), await counter.getAddress(), resetData, {
    nonce: 1n
  })
  await expect(
    entryPoint.validateAndExecute(badOp, await signOperation(entryPoint, badOp, sessionKey))
  ).to.be.revertedWithCustomError(entryPoint, 'InvalidSignature')
})

练习 2：给 paymaster 增加每日总预算与边界测试

在 MiniPaymaster.sol 中增加日预算字段，并在 sponsor 里按自然日滚动：

solidity
uint256 public immutable dailyBudget;
uint256 public dayStart;
uint256 public spentToday;

error DailyBudgetExceeded();

constructor(address entryPoint_, uint256 maxPerAccount_, uint256 dailyBudget_) {
    entryPoint = entryPoint_;
    maxPerAccount = maxPerAccount_;
    dailyBudget = dailyBudget_;
    dayStart = block.timestamp;
}

function sponsor(address account, uint256 amount) external onlyEntryPoint {
    if (block.timestamp >= dayStart + 1 days) {
        dayStart = block.timestamp;
        spentToday = 0;
    }

    if (spent[account] + amount > maxPerAccount) revert SponsorLimitExceeded();
    if (spentToday + amount > dailyBudget) revert DailyBudgetExceeded();
    if (depositBalance < amount) revert InsufficientDeposit();

    spent[account] += amount;
    spentToday += amount;
    depositBalance -= amount;
    emit Sponsored(account, amount);
}

边界测试（使用 impersonated EntryPoint 直接调用 sponsor，只测预算逻辑）：

ts
import { ethers, network } from 'hardhat'

it('每日总预算到达上限后拒绝，跨日后恢复', async () => {
  const { entryPoint, owner, other } = await loadFixture(deployFixture)
  const paymaster = await ethers.deployContract('MiniPaymaster', [
    await entryPoint.getAddress(),
    500n,
    600n
  ])
  await paymaster.deposit({ value: 1000n })

  const entryPointAddress = await entryPoint.getAddress()
  await network.provider.send('hardhat_impersonateAccount', [entryPointAddress])
  await network.provider.send('hardhat_setBalance', [
    entryPointAddress,
    '0x1000000000000000000'
  ])
  const impersonated = await ethers.getSigner(entryPointAddress)

  await paymaster.connect(impersonated).sponsor(owner.address, 500n)
  await paymaster.connect(impersonated).sponsor(other.address, 100n)

  await expect(
    paymaster.connect(impersonated).sponsor(other.address, 1n)
  ).to.be.revertedWithCustomError(paymaster, 'DailyBudgetExceeded')

  await time.increase(86_400)
  await paymaster.connect(impersonated).sponsor(other.address, 1n)
  expect(await paymaster.spentToday()).to.equal(1n)
})

练习 3：为账户增加社交恢复（guardian 投票更换 owner）

在 SmartAccount.sol 中加入 guardian 集合与 2-of-3 恢复流程。教学版把阈值固定为 2，生产环境应可配置并加 timelock。

solidity
mapping(address => bool) public guardians;
mapping(address => uint256) public recoveryApprovalRound;

uint256 public recoveryRound;
uint256 public recoveryApprovals;
address public pendingOwner;
uint256 public recoveryDeadline;

event GuardianUpdated(address indexed guardian, bool enabled);
event RecoveryProposed(address indexed newOwner, uint256 deadline);
event RecoveryApproved(address indexed guardian, uint256 approvals);
event OwnerRecovered(address indexed oldOwner, address indexed newOwner);

error NotGuardian();
error InvalidRecovery();
error RecoveryExpired();
error AlreadyRecoveryApproved();
error ZeroAddress();

function setGuardian(address guardian, bool enabled) external onlyOwner {
    if (guardian == address(0)) revert ZeroAddress();
    guardians[guardian] = enabled;
    emit GuardianUpdated(guardian, enabled);
}

function proposeRecovery(address newOwner) external {
    if (!guardians[msg.sender]) revert NotGuardian();
    if (newOwner == address(0) || newOwner == owner) revert InvalidRecovery();

    recoveryRound += 1;
    recoveryApprovals = 0;
    pendingOwner = newOwner;
    recoveryDeadline = block.timestamp + 7 days;

    emit RecoveryProposed(newOwner, recoveryDeadline);
}

function approveRecovery() external {
    if (!guardians[msg.sender]) revert NotGuardian();
    if (pendingOwner == address(0) || block.timestamp > recoveryDeadline) {
        revert RecoveryExpired();
    }
    if (recoveryApprovalRound[msg.sender] == recoveryRound) revert AlreadyRecoveryApproved();

    recoveryApprovalRound[msg.sender] = recoveryRound;
    recoveryApprovals += 1;
    emit RecoveryApproved(msg.sender, recoveryApprovals);

    if (recoveryApprovals >= 2) {
        address oldOwner = owner;
        owner = pendingOwner;
        pendingOwner = address(0);
        recoveryApprovals = 0;
        emit OwnerRecovered(oldOwner, owner);
    }
}

练习 4：模拟 bundler 批量提交多个 UserOperation

nonce 必须严格递增，因此 bundler 不能随意打乱顺序。把下面测试加到 test/AccountAbstraction.test.ts：

ts
it('bundler 必须按 nonce 顺序提交', async () => {
  const { entryPoint, account, counter, owner } = await loadFixture(deployFixture)
  const accountAddress = await account.getAddress()
  const counterAddress = await counter.getAddress()
  const data = counter.interface.encodeFunctionData('increment')

  const ops = []
  for (let nonce = 0n; nonce < 3n; nonce++) {
    ops.push(await buildOperation(accountAddress, counterAddress, data, { nonce }))
  }

  // 先提交 nonce=1 会被拒绝
  const outOfOrder = await signOperation(entryPoint, ops[1], owner)
  await expect(entryPoint.validateAndExecute(ops[1], outOfOrder)).to.be.revertedWithCustomError(
    entryPoint,
    'InvalidNonce'
  )

  for (const op of ops) {
    const signature = await signOperation(entryPoint, op, owner)
    await entryPoint.validateAndExecute(op, signature)
  }

  expect(await counter.value()).to.equal(3n)
  expect(await entryPoint.nonces(accountAddress)).to.equal(3n)
})

练习 5：对比真实 EntryPoint 的 validateUserOp，列出教学版缺失的检查

教学版 hashOperation 只编码了 8 个字段，validateUserOp 也只返回 bool。与真实 ERC-4337 相比，至少缺少：

缺失项风险
chainId 与 EntryPoint 地址未进入 userOpHash同一签名可在其他链或另一个 EntryPoint 部署上重放
二维 nonce（key + sequence）无法按 key 分域管理 nonce，只支持单一路径
verificationGasLimit / preVerificationGas / paymasterData无法正确计量验证成本，paymaster 也无法传策略参数
validAfter / validUntil不支持时间窗，只能靠 session key 的 expiresAt
aggregator 与聚合签名不支持多签聚合，验证接口过窄
返回 validationData 而非 bool无法把签名失败、时间窗、聚合器信息统一交给 EntryPoint
paymaster 的 EntryPoint 存款模型本示例是合约内记账，没有真实托管、提款与退款
postOp 回调与失败回滚paymaster 无法处理执行后的结算与惩罚
验证阶段的存储访问限制真实实现要限制验证阶段可访问的存储，降低 DoS 面
signature 的 ABI 编码教学版单独传 bytes，真实结构把它作为 UserOperation 字段

最关键的一条是 userOpHash 缺 chainId 与 EntryPoint 地址。真实实现通常做类似：

text
userOpHash = keccak256(abi.encode(keccak256(abi.encode(op)), chainId, entryPoint))
1

修复方向是先把域分隔加进哈希，再补齐 gas 字段、时间窗与 paymaster 存款模型。结论仍然只适用于教学实现，生产请使用经过审计的实现与成熟钱包 SDK。
