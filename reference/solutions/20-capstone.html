solidity
function cancelCampaign(uint256 id) external {
    Campaign storage campaign = campaigns[id];
    if (campaign.creator == address(0)) revert CampaignNotFound();
    if (msg.sender != campaign.creator) revert NotCreator();
    if (campaign.finalized) revert AlreadyFinalized();
    if (campaign.cancelled) revert AlreadyCancelled();
    if (block.timestamp >= campaign.deadline) revert DeadlinePassed();
    if (campaign.pledged != 0) revert CannotCancel();

    campaign.cancelled = true;
    emit CampaignCancelled(id, msg.sender);
}

练习 2：增加最低认缴额与单地址上限，并说明对退款的影响

把两个限制放进 Campaign，由创建者设定：

solidity
struct Campaign {
    // ... 原有字段
    uint256 minPledge;
    uint256 maxPerAddress; // 0 表示不限制
}

error InvalidPledgeLimits();
error BelowMinPledge();
error AddressCapExceeded();

function createCampaign(
    uint256 goal,
    uint256 deadline,
    uint256 minPledge,
    uint256 maxPerAddress
) external returns (uint256 id) {
    if (goal == 0) revert InvalidGoal();
    if (deadline <= block.timestamp) revert InvalidDeadline();
    if (minPledge == 0) revert InvalidPledgeLimits();
    if (maxPerAddress != 0 && maxPerAddress < minPledge) revert InvalidPledgeLimits();

    id = nextId++;
    campaigns[id] = Campaign(
        msg.sender, goal, deadline, 0, false, false, minPledge, maxPerAddress
    );
    emit CampaignCreated(id, msg.sender, goal, deadline);
}

这会改动 createCampaign 的签名，现有测试里的 createGoalCampaign helper 和所有创建调用都要同步补上 minPledge 与 maxPerAddress（不需要限制时传 1n 与 0n）。如果同时做了练习 1，结构体初始化还要多一个 false 给 cancelled。

对退款的影响：

  • 退款仍然按 contributions[id][msg.sender] 单点读取，不需要遍历每一笔认缴。
  • 上限限制的是“同一地址累计总额”，不是单笔金额；多次认缴累加后仍不能超过上限。
  • 退款金额不受上限影响，因为上限在认缴时已经约束了累计值。
  • 如果产品要求“降低上限后追回超额部分”，需要额外管理函数与退款路径；教学版不做这种动态调整。
练习 3：把 finalize 改成惰性结算（首次提取/退款时触发）

抽出一个内部函数，避免在 withdraw 与 refund 里复制状态判断：

solidity
function _finalize(uint256 id, Campaign storage campaign) internal {
    if (campaign.finalized) return;
    if (block.timestamp < campaign.deadline) revert DeadlineNotReached();

    campaign.finalized = true;
    emit Finalized(id, campaign.pledged >= campaign.goal);
}

withdraw 改为先惰性结算，再检查目标：

solidity
function withdraw(uint256 id) external nonReentrant {
    Campaign storage campaign = campaigns[id];
    if (campaign.creator == address(0)) revert CampaignNotFound();
    if (msg.sender != campaign.creator) revert NotCreator();
    if (campaign.cancelled) revert AlreadyCancelled();

    _finalize(id, campaign);

    if (campaign.pledged < campaign.goal) revert GoalNotReached();
    if (campaign.withdrawn) revert AlreadyWithdrawn();

    uint256 amount = campaign.pledged;
    campaign.withdrawn = true;

    (bool ok, ) = msg.sender.call{ value: amount }('');
    if (!ok) revert CallFailed();

    emit Withdrawn(id, msg.sender, amount);
}

refund 同理：

solidity
function refund(uint256 id) external nonReentrant {
    Campaign storage campaign = campaigns[id];
    if (campaign.creator == address(0)) revert CampaignNotFound();
    if (campaign.cancelled) revert AlreadyCancelled();

    _finalize(id, campaign);

    if (campaign.pledged >= campaign.goal) revert GoalReached();
    if (refunded[id][msg.sender]) revert AlreadyRefunded();

    uint256 amount = contributions[id][msg.sender];
    if (amount == 0) revert NothingToRefund();

    refunded[id][msg.sender] = true;

    (bool ok, ) = msg.sender.call{ value: amount }('');
    if (!ok) revert CallFailed();

    emit Refunded(id, msg.sender, amount);
}

两种设计对比：

维度显式 finalize惰性结算
交易数量多一笔结算交易少一笔，首次领取即结算
状态可读性到期后状态及时更新无人领取时链上 finalized 一直是 false
索引/前端逻辑简单必须同时看 finalized 与 deadline，或监听 Finalized
一致性所有领取路径都依赖已结算_finalize 必须在 withdraw/refund 都调用，漏一处就不一致
失败回滚结算独立成功领取失败会连带回滚结算事件

惰性结算适合降低用户操作步骤，但前端与索引器要能处理“已到期但未结算”的中间状态。

练习 4：写一个 SQLite 索引器，支持按活动 ID 查询时间线

新建 examples/20-capstone/scripts/indexer.mjs（本练习新增）。使用 Node 内置的 node:sqlite，不需要新增依赖（Node 22.5+ 可用，本环境 Node 24）：

js
import { readFileSync } from 'node:fs'

import { DatabaseSync } from 'node:sqlite'
import { Interface, JsonRpcProvider } from 'ethers'

const deployment = JSON.parse(readFileSync('deployments/31337.json', 'utf8'))
const provider = new JsonRpcProvider('http://127.0.0.1:8545')
const iface = new Interface(deployment.abi)
const db = new DatabaseSync('launchpad.db')

db.exec(`
  CREATE TABLE IF NOT EXISTS events (
    block_number INTEGER NOT NULL,
    log_index INTEGER NOT NULL,
    tx_hash TEXT NOT NULL,
    name TEXT NOT NULL,
    campaign_id TEXT,
    args TEXT NOT NULL,
    PRIMARY KEY (tx_hash, log_index)
  );
`)

const logs = await provider.getLogs({
  address: deployment.address,
  fromBlock: 0,
  toBlock: 'latest'
})

const insert = db.prepare(
  'INSERT OR IGNORE INTO events VALUES (?, ?, ?, ?, ?, ?)'
)

for (const log of logs) {
  const parsed = iface.parseLog(log)
  if (!parsed) continue

  const args = Object.fromEntries(
    Object.entries(parsed.args.toObject()).map(([key, value]) => [
      key,
      typeof value === 'bigint' ? value.toString() : value
    ])
  )

  insert.run(
    log.blockNumber,
    log.index,
    log.transactionHash,
    parsed.name,
    parsed.args.id === undefined ? null : String(parsed.args.id),
    JSON.stringify(args)
  )
}

const campaignId = process.argv[2] ?? '0'
const rows = db
  .prepare(
    `SELECT block_number, log_index, name, args
     FROM events
     WHERE campaign_id = ?
     ORDER BY block_number, log_index`
  )
  .all(campaignId)

console.log(rows)

练习 5：给前端做一个只读页面

新建 examples/20-capstone/frontend/read.html（本练习新增）。页面只读，不请求签名：

html
<!doctype html>
<html lang="zh-CN">
<head>
  <meta charset="utf-8" />
  <title>MiniLaunchpad 只读查询</title>
</head>
<body>
  <h1>MiniLaunchpad 只读查询</h1>

  <p>
    <label>合约地址 <input id="address" size="52" placeholder="0x..." /></label>
  </p>
  <p>
    <label>活动 ID <input id="campaignId" size="6" value="0" /></label>
  </p>
  <button id="load">查询</button>
  <pre id="out">填写地址后点击查询</pre>

  <script src="https://cdn.jsdelivr.net/npm/ethers@6.13.4/dist/ethers.umd.min.js"></script>
  <script>
    const ABI = [
      'function campaigns(uint256) view returns (address creator, uint256 goal, uint256 deadline, uint256 pledged, bool finalized, bool withdrawn)',
      'event CampaignCreated(uint256 indexed id, address indexed creator, uint256 goal, uint256 deadline)',
      'event Pledged(uint256 indexed id, address indexed contributor, uint256 amount)',
      'event Finalized(uint256 indexed id, bool success)',
      'event Withdrawn(uint256 indexed id, address indexed creator, uint256 amount)',
      'event Refunded(uint256 indexed id, address indexed contributor, uint256 amount)'
    ]

    const provider = new ethers.JsonRpcProvider('http://127.0.0.1:8545')

    function statusOf(campaign, now) {
      if (campaign.withdrawn) return '已提取'
      if (campaign.finalized) {
        return campaign.pledged >= campaign.goal ? '成功，可提取' : '未达标，可退款'
      }
      return now >= campaign.deadline ? '待结算' : '募集中'
    }

    document.getElementById('load').onclick = async () => {
      const out = document.getElementById('out')
      out.textContent = '查询中...'

      try {
        const address = document.getElementById('address').value.trim()
        const id = BigInt(document.getElementById('campaignId').value || '0')
        const contract = new ethers.Contract(address, ABI, provider)

        const campaign = await contract.campaigns(id)
        const now = BigInt(Math.floor(Date.now() / 1000))
        const status = statusOf(campaign, now)

        const logs = await provider.getLogs({
          address,
          fromBlock: 0,
          toBlock: 'latest'
        })

        const timeline = logs
          .map((log) => contract.interface.parseLog(log))
          .filter((event) => event && event.args.id === id)
          .map((event) => {
            const args = event.args.toObject()
            for (const key of Object.keys(args)) {
              if (typeof args[key] === 'bigint') args[key] = args[key].toString()
            }
            return { name: event.name, args }
          })

        out.textContent = JSON.stringify(
          {
            creator: campaign.creator,
            goalEth: ethers.formatEther(campaign.goal),
            pledgedEth: ethers.formatEther(campaign.pledged),
            deadline: Number(campaign.deadline),
            status,
            timeline
          },
          null,
          2
        )
      } catch (error) {
        out.textContent = `查询失败: ${error.message}`
      }
    }
  </script>
</body>
</html>

练习 6：写一份一页纸的上线检查表
领域检查项证据/负责人
合约状态机覆盖创建、认缴、结算、提取、退款、取消状态图 + 测试清单

合约重复提取、重复退款、非创建者调用都有测试pnpm --filter @web3-ex/20-capstone test 输出
合约自定义错误与事件覆盖关键状态ABI 与事件表
测试正常路径、边界、权限、余额守恒13 条现有测试 + 新增用例
测试达标与未达标两条资金流都验证changeEtherBalances 断言
部署按 chainId 写地址簿，记录部署块与构造参数deployments/<chainId>.json
部署部署后跑冒烟脚本，合约余额归零pnpm --filter @web3-ex/20-capstone demo
索引从部署块开始，不漏 CampaignCreated索引器测试
索引BigInt 转字符串、主键去重、重组处理数据库 schema 与运行记录
前端只读优先，写操作走钱包确认页面评审
前端错误选择器有映射，地址不写死错误映射表
安全已知限制、中心化权限、升级方式、应急联系人交付说明
应急发现异常时谁能暂停、如何通知、如何回滚runbook + 演练记录

上线前最后确认：

  • 合约没有默认可见的 receive 资金黑洞或未授权提取路径。
  • 演示脚本的“合约余额 0.0 ETH”结论只适用于完整流程；部分退款场景也要单独检查守恒。
  • 所有地址来自地址簿，不把本地链地址硬编码进前端。
  • 交付说明里明确写“教学实现，未审计，不处理真实资金”。
