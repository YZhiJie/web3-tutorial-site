本章目标 ​

完成这一章后，你应该能够：

  • 把一个产品需求拆成合约状态机、外部接口与事件模型。
  • 实现“创建 → 认缴 → 结算 → 提取/退款”的完整资金流程。
  • 用状态位与 nonReentrant 同时覆盖重复调用与重入两类风险。
  • 写出可复现的测试、部署脚本与端到端演练脚本。
  • 用 eth_getLogs 重建事件时间线，并做余额守恒检查。
  • 输出一份包含已知限制与上线前检查项的交付说明。

前置检查 ​

  • 已完成 15 至 19 五章。
  • 能独立写 Hardhat 测试、部署脚本与最小索引脚本。
  • 理解 CEI、自定义错误、事件与访问控制。

1. 需求与状态机 ​

产品需求只有一句话：发起人设定目标金额与截止时间，支持者认缴；到期后达标则发起人提取，未达标则支持者退款。

把它落成状态机：

text
创建（creator, goal, deadline）
   ↓ pledge（截止前，可多次）
认缴中
   ↓ finalize（截止后，任何人）
已结算（success = pledged >= goal）
   ├─ success → withdraw（仅 creator，一次）
   └─ fail    → refund（每位 contributor，一次）

三个不变量：

  1. 截止后不能再认缴。
  2. 结算只能发生一次。
  3. 每笔认缴最多被提取或退款一次。

2. 状态与事件模型 ​

solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

/// @title MiniLaunchpad
/// @notice Capstone 合约：创建募资、认缴、结算、提取或退款。
/// @dev 教学实现，不含白名单、KYC、代币发行与前端。
// #region campaign-state
contract MiniLaunchpad {
    struct Campaign {
        address creator;
        uint256 goal;
        uint256 deadline;
        uint256 pledged;
        bool finalized;
        bool withdrawn;
    }

    mapping(uint256 => Campaign) public campaigns;
    mapping(uint256 => mapping(address => uint256)) public contributions;
    mapping(uint256 => mapping(address => bool)) public refunded;

    uint256 public nextId;
    uint256 private locked;

    event CampaignCreated(uint256 indexed id, address indexed creator, uint256 goal, uint256 deadline);
    event Pledged(uint256 indexed id, address indexed contributor, uint256 amount);
    event Finalized(uint256 indexed id, bool success);
    event Withdrawn(uint256 indexed id, address indexed creator, uint256 amount);
    event Refunded(uint256 indexed id, address indexed contributor, uint256 amount);

    error CampaignNotFound();
    error InvalidDeadline();
    error InvalidGoal();
    error ZeroAmount();
    error AlreadyFinalized();
    error NotFinalized();
    error DeadlineNotReached();
    error DeadlinePassed();
    error GoalNotReached();
    error GoalReached();
    error NotCreator();
    error AlreadyWithdrawn();
    error NothingToRefund();
    error AlreadyRefunded();
    error ReentrantCall();
    error CallFailed();

    modifier nonReentrant() {
        if (locked == 1) revert ReentrantCall();
        locked = 1;
        _;
        locked = 0;
    }
    // #endregion

    // #region create-pledge
    function createCampaign(uint256 goal, uint256 deadline) external returns (uint256 id) {
        if (goal == 0) revert InvalidGoal();
        if (deadline <= block.timestamp) revert InvalidDeadline();

        id = nextId++;
        campaigns[id] = Campaign(msg.sender, goal, deadline, 0, false, false);
        emit CampaignCreated(id, msg.sender, goal, deadline);
    }

    function pledge(uint256 id) external payable {
        Campaign storage campaign = campaigns[id];
        if (campaign.creator == address(0)) revert CampaignNotFound();
        if (campaign.finalized) revert AlreadyFinalized();
        if (block.timestamp >= campaign.deadline) revert DeadlinePassed();
        if (msg.value == 0) revert ZeroAmount();

        campaign.pledged += msg.value;
        contributions[id][msg.sender] += msg.value;
        emit Pledged(id, msg.sender, msg.value);
    }
    // #endregion

    // #region finalize-settle
    function finalize(uint256 id) external {
        Campaign storage campaign = campaigns[id];
        if (campaign.creator == address(0)) revert CampaignNotFound();
        if (campaign.finalized) revert AlreadyFinalized();
        if (block.timestamp < campaign.deadline) revert DeadlineNotReached();

        campaign.finalized = true;
        emit Finalized(id, campaign.pledged >= campaign.goal);
    }

    function withdraw(uint256 id) external nonReentrant {
        Campaign storage campaign = campaigns[id];
        if (msg.sender != campaign.creator) revert NotCreator();
        if (!campaign.finalized) revert NotFinalized();
        if (campaign.pledged < campaign.goal) revert GoalNotReached();
        if (campaign.withdrawn) revert AlreadyWithdrawn();

        uint256 amount = campaign.pledged;
        campaign.withdrawn = true;

        (bool ok, ) = msg.sender.call{ value: amount }('');
        if (!ok) revert CallFailed();

        emit Withdrawn(id, msg.sender, amount);
    }

    function refund(uint256 id) external nonReentrant {
        Campaign storage campaign = campaigns[id];
        if (!campaign.finalized) revert NotFinalized();
        if (campaign.pledged >= campaign.goal) revert GoalReached();
        if (refunded[id][msg.sender]) revert AlreadyRefunded();

        uint256 amount = contributions[id][msg.sender];
        if (amount == 0) revert NothingToRefund();

        refunded[id][msg.sender] = true;

        (bool ok, ) = msg.sender.call{ value: amount }('');
        if (!ok) revert CallFailed();

        emit Refunded(id, msg.sender, amount);
    }
    // #endregion
}

设计要点：

  • campaigns 存活动主状态，contributions 存每人对每个活动的金额。
  • refunded[id][account] 与 campaign.withdrawn 是防重复的显式状态位。
  • locked 是重入锁，和状态位互补：状态位防业务重复，锁防调用期间重入。
  • 每个状态变化都发事件，链下索引不依赖内部存储布局。

边界检查：

  • goal == 0 与 deadline <= now 在创建时拒绝，避免产生永远无法结算的活动。
  • pledge 同时检查活动存在、未结算、未截止、金额非零。
  • pledged 与 contributions 同步累加，保证总额与个人份额一致。

为什么用增量累加

增量写入只触碰目标 slot，避免每次认缴都遍历全部贡献者。退款时再按 contributions[id][msg.sender] 单点读取。

资金流出遵循同一套顺序：

  1. 校验调用者与状态。
  2. 先把 withdrawn / refunded 置位。
  3. 再执行外部 call。
  4. 失败则整体回滚，成功才发事件。

withdraw 与 refund 都带 nonReentrant，即使未来加入回调型接收者，也不会在状态更新前重入。

不要用 transfer 发资金

transfer 的固定 gas 上限会拒绝复杂的合约接收者。使用 call 并检查返回值，同时配合 CEI 与重入锁。

6. 部署与地址簿 ​

typescript
import { mkdirSync, writeFileSync } from 'node:fs'
import { dirname, resolve } from 'node:path'

import { artifacts, ethers } from 'hardhat'

async function main() {
  // #region deploy-script
  const launchpad = await ethers.deployContract('MiniLaunchpad')
  await launchpad.waitForDeployment()

  const address = await launchpad.getAddress()
  const chain = await ethers.provider.getNetwork()
  const artifact = await artifacts.readArtifact('MiniLaunchpad')

  const outputPath = resolve(process.cwd(), 'deployments', `${chain.chainId}.json`)
  mkdirSync(dirname(outputPath), { recursive: true })
  writeFileSync(
    outputPath,
    JSON.stringify(
      {
        chainId: Number(chain.chainId),
        address,
        abi: artifact.abi,
        deployedAt: new Date().toISOString()
      },
      null,
      2
    )
  )

  console.log(`MiniLaunchpad deployed to ${address} on chain ${chain.chainId}`)
  console.log(`地址簿已写入 ${outputPath}`)
  // #endregion
}

main().catch((error) => {
  console.error(error)
  process.exitCode = 1
})

7. 端到端演练与事件索引 ​

bash
pnpm --filter @web3-ex/20-capstone demo

脚本在本地链上完成：创建目标 3 ETH 的活动 → Alice 认缴 2 ETH → Bob 认缴 1.5 ETH → 推进时间 → 结算 → 创建者提取 3.5 ETH。

随后用 eth_getLogs 拉取部署块到最新块的全部日志：

typescript
import { ethers, network } from 'hardhat'
import { time } from '@nomicfoundation/hardhat-toolbox/network-helpers'

async function main() {
  const [creator, alice, bob] = await ethers.getSigners()
  const launchpad = await ethers.deployContract('MiniLaunchpad')
  await launchpad.waitForDeployment()

  const address = await launchpad.getAddress()
  const deployedBlock = (await launchpad.deploymentTransaction())!.blockNumber
  const deadline = (await time.latest()) + 7200

  console.log('1. 创建活动，目标 3 ETH')
  await launchpad.connect(creator).createCampaign(ethers.parseEther('3'), deadline)

  console.log('2. Alice 认缴 2 ETH，Bob 认缴 1.5 ETH')
  await launchpad.connect(alice).pledge(0, { value: ethers.parseEther('2') })
  await launchpad.connect(bob).pledge(0, { value: ethers.parseEther('1.5') })

  console.log('3. 推进到截止时间并结算')
  await time.increaseTo(deadline)
  await launchpad.finalize(0)

  console.log('4. 创建者提取 3.5 ETH')
  await launchpad.connect(creator).withdraw(0)

  // #region e2e-index
  const logs = await ethers.provider.getLogs({
    address,
    fromBlock: deployedBlock,
    toBlock: 'latest'
  })

  const events = logs.map((log) => {
    const parsed = launchpad.interface.parseLog(log)
    return {
      blockNumber: log.blockNumber,
      transactionHash: log.transactionHash,
      name: parsed?.name,
      args: parsed?.args.toObject()
    }
  })

  const summary = events.map((event) => ({
    ...event,
    args: Object.fromEntries(
      Object.entries(event.args ?? {}).map(([key, value]) => [
        key,
        typeof value === 'bigint' ? value.toString() : value
      ])
    )
  }))
  // #endregion

  console.log('5. 索引结果')
  console.log(JSON.stringify(summary, null, 2))

}

事件参数里的金额是 bigint，JSON.stringify 会抛错。脚本统一转成字符串后再输出，索引器落库时也应保持十进制字符串或最小单位整数。

8. 交付清单 ​

上线前至少确认：

  • 合约：状态机、事件、自定义错误、重入防护、资金守恒。
  • 测试：正常路径、边界、重复调用、权限与余额断言。
  • 部署：按链 ID 的地址簿、部署区块号、构造参数记录。
  • 索引：起始块、事件解码、重组处理、失败重试。
  • 前端：只读优先、写操作走钱包确认、错误选择器映射。
  • 安全说明：已知限制、中心化权限、升级方式、应急联系人。

9. 常见坑 ​

  • 只测成功路径：没有覆盖重复提取、重复退款、非创建者调用。
  • 先转账后改状态：给重入留下窗口。
  • 结算与提取混在一个函数：权限与状态边界会变得模糊。
  • 索引从最新块开始：会漏掉部署与早期事件。
  • 把地址写死在前端：本地链重启后全部失效。
  • 忽略 BigInt 序列化：索引落库或日志输出直接报错。
  • 没有守恒检查：资金去向无法快速验证。

10. 练习题 ​

  1. 增加“创建者取消活动”功能，要求截止前且无人认缴，并补测试。
  2. 增加最低认缴额与单地址上限，说明对退款逻辑的影响。
  3. 把 finalize 改成自动触发（例如首次提取时惰性结算），比较两种设计的优缺点。
  4. 写一个索引器，把事件写入 SQLite，并支持按活动 ID 查询时间线。
  5. 给前端做一个只读页面：输入地址与活动 ID，展示目标、已认缴、状态与事件。
  6. 写一份一页纸的上线检查表，覆盖合约、测试、部署、索引、前端与应急。
