16 · 账户抽象（ERC-4337） ​

需进阶约 6 学时 前置：/advanced/07-solidity-patterns、/advanced/11-token-standards验证于 2026-11-04

本章目标 ​

完成这一章后，你应该能够：

  • 解释 EOA 的限制，以及账户抽象要解决的问题。
  • 描述 UserOperation 从构造、签名、验证到执行的完整流程。
  • 用 nonce 与已处理状态防止重放。
  • 设计受限 session key：绑定目标、有效期，并支持撤销。
  • 说明 paymaster 的预存、限额与赞助失败路径。
  • 区分教学实现与真实 ERC-4337 的差距。

前置检查 ​

  • 已完成 07 · Solidity 进阶与合约模式 与 11 · Token 标准。
  • 理解 ecrecover、EIP-191 签名前缀与 nonce。
  • 知道 delegatecall 与外部调用的基本风险。

本章示例位于 examples/16-account-abstraction，包含 5 个合约与 11 条测试，全部在 Hardhat Network 运行。

1. EOA 的三条硬约束 ​

普通 EOA 只能：

  • 用私钥签名交易，无法自定义验证逻辑。
  • 用私钥直接支付 gas，无法由第三方代付。
  • 丢失私钥即失去账户，无法设置恢复人或限额。

账户抽象把“账户”变成合约：验证逻辑、gas 支付、权限策略都可以写成代码。

2. UserOperation 是什么 ​

ERC-4337 不直接发交易，而是提交 UserOperation。教学版结构：

solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

// #region user-op
struct UserOperation {
    address account;
    uint256 nonce;
    address target;
    uint256 value;
    bytes data;
    address paymaster;
    uint256 maxFeePerGas;
    uint256 callGasLimit;
}

interface IEntryPoint {
    function hashOperation(UserOperation calldata op) external view returns (bytes32);

interface ISmartAccount {
    function execute(address target, uint256 value, bytes calldata data) external;

    function validateUserOp(
        UserOperation calldata op,
        bytes calldata signature
    ) external view returns (bool);
}

interface IPaymaster {
    function sponsor(address account, uint256 amount) external;
}

端点实现：

solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import { UserOperation, IEntryPoint, ISmartAccount, IPaymaster } from './IEntryPoint.sol';

/// @title MiniEntryPoint
/// @notice 教学用最小 EntryPoint：校验 nonce、签名与 paymaster，然后执行调用。
/// @dev 不是 ERC-4337 的完整实现，缺少 bundler 经济学、聚合签名与 UserOperation 打包。
// #region entrypoint
contract MiniEntryPoint is IEntryPoint {
    mapping(address => uint256) public nonces;

    event UserOperationExecuted(
        address indexed account,
        uint256 indexed nonce,
        address target,
        uint256 value
    );

    error InvalidNonce();
    error InvalidSignature();
    error CallFailed();

    function hashOperation(
        UserOperation calldata op
    ) public pure override returns (bytes32) {
        return
            keccak256(
                abi.encode(
                    op.account,
                    op.nonce,
                    op.target,
                    op.value,
                    keccak256(op.data),
                    op.paymaster,
                    op.maxFeePerGas,
                    op.callGasLimit
                )
            );
    }

    function validateAndExecute(
        UserOperation calldata op,
        bytes calldata signature
    ) external {
        if (op.nonce != nonces[op.account]) revert InvalidNonce();

        if (!ISmartAccount(op.account).validateUserOp(op, signature)) {
            revert InvalidSignature();
        }

        // 先递增 nonce，再做外部调用，避免重入时复用同一个 nonce
        nonces[op.account] = op.nonce + 1;

        if (op.paymaster != address(0)) {
            IPaymaster(op.paymaster).sponsor(
                op.account,
                op.callGasLimit * op.maxFeePerGas
            );
        }

        (bool ok, ) = op.account.call(
            abi.encodeCall(ISmartAccount.execute, (op.target, op.value, op.data))
        );
        if (!ok) revert CallFailed();

        emit UserOperationExecuted(op.account, op.nonce, op.target, op.value);
    }
}
// #endregion

4. 智能账户：owner 与 session key ​

solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import { UserOperation, IEntryPoint, ISmartAccount } from './IEntryPoint.sol';

/// @title SmartAccount
/// @notice 由 owner 或受限 session key 控制的最小智能账户。
// #region account
contract SmartAccount is ISmartAccount {
    struct SessionKey {
        address allowedTarget;
        uint256 expiresAt;
    }

    IEntryPoint public immutable entryPoint;
    address public owner;
    mapping(address => SessionKey) public sessionKeys;

    event Executed(address indexed target, uint256 value, bytes data);
    event SessionKeyAdded(address indexed key, address indexed allowedTarget, uint256 expiresAt);

    error NotEntryPoint();
    error NotOwner();
    error InvalidSignature();
    error SessionKeyExpired();
    error TargetNotAllowed();
    error InvalidSessionKey();

    modifier onlyEntryPoint() {
        if (msg.sender != address(entryPoint)) revert NotEntryPoint();
        _;
    }

    modifier onlyOwner() {
        if (msg.sender != owner) revert NotOwner();
        _;
    }

    constructor(address entryPoint_, address owner_) {
        entryPoint = IEntryPoint(entryPoint_);
        owner = owner_;
    }

    // #region session-key
    function addSessionKey(
        address key,
        address allowedTarget,
        uint256 expiresAt
    ) external onlyOwner {
        if (key == address(0) || allowedTarget == address(0) || expiresAt <= block.timestamp) {
            revert InvalidSessionKey();
        }
        sessionKeys[key] = SessionKey(allowedTarget, expiresAt);
        emit SessionKeyAdded(key, allowedTarget, expiresAt);
    }

    function revokeSessionKey(address key) external onlyOwner {
        delete sessionKeys[key];
    }
    // #endregion

    // #region account-validate
    function validateUserOp(
        UserOperation calldata op,
        bytes calldata signature
    ) external view override returns (bool) {
        if (op.account != address(this)) return false;

        bytes32 hash = entryPoint.hashOperation(op);
        address signer = _recover(hash, signature);

        if (signer == owner) return true;

        SessionKey memory key = sessionKeys[signer];
        if (key.expiresAt == 0) return false;
        if (block.timestamp > key.expiresAt) return false;
        if (op.target != key.allowedTarget) return false;

        return true;
    }
    // #endregion

    function execute(
        address target,
        uint256 value,
        bytes calldata data
    ) external onlyEntryPoint {
        (bool ok, ) = target.call{ value: value }(data);
        require(ok, 'SmartAccount: call failed');
        emit Executed(target, value, data);
    }

    function _recover(bytes32 hash, bytes calldata signature) internal pure returns (address) {
        if (signature.length != 65) return address(0);

        bytes32 r;
        bytes32 s;
        uint8 v;
        assembly {
            r := calldataload(signature.offset)
            s := calldataload(add(signature.offset, 32))
            v := byte(0, calldataload(add(signature.offset, 64)))
        }
        if (v < 27) v += 27;

        return
            ecrecover(
                keccak256(abi.encodePacked('\x19Ethereum Signed Message:\n32', hash)),
                v,
                r,
                s
            );
    }

    receive() external payable {}
}
// #endregion

教学版 session key 只限制两件事：

  • 目标地址：只能调用预先允许的合约。
  • 有效期：超过 expiresAt 自动失效。

生产环境还应限制：

  • 函数选择器与参数范围。
  • 单笔与累计金额上限。
  • 可撤销、可轮换，并有链下告警。

6. Paymaster：谁付 gas ​

solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import { IPaymaster } from './IEntryPoint.sol';

/// @title MiniPaymaster
/// @notice 教学用 paymaster：预存额度，按账户限额为 UserOperation 付费。
// #region paymaster
contract MiniPaymaster is IPaymaster {
    address public immutable entryPoint;
    uint256 public immutable maxPerAccount;

    uint256 public depositBalance;
    mapping(address => uint256) public spent;

    event Deposited(address indexed from, uint256 amount);
    event Sponsored(address indexed account, uint256 amount);

    error NotEntryPoint();
    error SponsorLimitExceeded();
    error InsufficientDeposit();

    modifier onlyEntryPoint() {
        if (msg.sender != entryPoint) revert NotEntryPoint();
        _;
    }

    constructor(address entryPoint_, uint256 maxPerAccount_) {
        entryPoint = entryPoint_;
        maxPerAccount = maxPerAccount_;
    }

    function deposit() external payable {
        depositBalance += msg.value;
        emit Deposited(msg.sender, msg.value);
    }

    function sponsor(address account, uint256 amount) external onlyEntryPoint {
        if (spent[account] + amount > maxPerAccount) revert SponsorLimitExceeded();
        if (depositBalance < amount) revert InsufficientDeposit();

        spent[account] += amount;
        depositBalance -= amount;
        emit Sponsored(account, amount);
    }
}
// #endregion

三个检查：

  1. 只有 EntryPoint 能调用 sponsor。
  2. 单账户累计不超过 maxPerAccount。
  3. 总预存足够支付本次费用。

真实 paymaster 还要处理：

  • 存款与提款（EntryPoint 托管）。
  • gas 价格上限与退款。
  • 赞助策略、风控与滥用防护。
  • 验证阶段的 postOp 回调与失败回滚。

7. 教学实现与真实 ERC-4337 的差距 ​

维度本示例真实 ERC-4337
UserOperation 字段精简完整 gas 字段与 paymasterData
bundler无链下服务，需模拟与打包
gas 计量简化verification / execution 分开计量
paymaster 资金合约内记账EntryPoint 存款模型
聚合签名无支持多签名聚合
账户恢复无社交恢复、guardian、模块化

不要直接用于生产

本示例用于理解流程与安全边界。生产请使用经过审计的 EntryPoint 实现与成熟钱包 SDK。

8. 安全清单 ​

  • nonce 必须由 EntryPoint 管理，账户不要自己维护一套。
  • 先验证签名，再消耗 nonce。
  • 先递增 nonce，再做外部调用。
  • execute 只能由 EntryPoint 调用。
  • session key 必须有目标、函数、金额、时间四类约束。
  • paymaster 必须有总预算与单账户限额。
  • 所有失败路径都要有明确错误，不能静默吞掉。
  • 升级账户实现时要走 timelock 或多签，避免单点接管。

  • owner 签名执行、事件与 nonce 递增。
  • 无效签名、错误 nonce、重放。
  • 账户只能由 EntryPoint 调用。
  • paymaster 扣费、限额拒绝。
  • session key 允许目标、拒绝其他目标、过期、撤销。

10. 常见坑 ​

  • 直接签裸哈希：前端 signMessage 会加 EIP-191 前缀，合约必须用同样的前缀恢复。
  • nonce 在验证前递增：签名失败也会消耗 nonce，用户体验很差。
  • session key 只绑地址：同一合约的不同函数可能权限差异巨大。
  • paymaster 不检查总额：单账户限额通过后仍可能被批量耗尽。
  • 忽略 execute 的返回值：外部调用失败必须回滚，不能继续发事件。
  • 把 owner 写成可变单点：没有多签或 timelock 的 owner 是最大的中心化风险。

11. 练习题 ​

  1. 给 session key 增加函数选择器白名单，并写越权测试。
  2. 给 paymaster 增加每日总预算，写边界测试。
  3. 为账户增加社交恢复：guardian 投票更换 owner。
  4. 模拟 bundler：批量提交多个 UserOperation，观察 nonce 顺序。
  5. 对比真实 EntryPoint 的 validateUserOp 字段，列出教学版缺失的安全检查。
