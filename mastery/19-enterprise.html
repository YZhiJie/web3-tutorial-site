19 · 企业级架构 ​

需进阶约 6 学时 前置：/mastery/18-audit、/advanced/09-deploy-verify验证于 2026-11-04

本章目标 ​

完成这一章后，你应该能够：

  • 设计“多签 + timelock + 金库 + guardian”的分层权限模型。
  • 实现 2-of-3 提案、确认、执行的完整流程，并防止重放。
  • 解释 timelock 的排队、eta、取消与执行边界。
  • 设计紧急暂停与解除暂停的非对称权限。
  • 列出监控事件、值班流程与升级治理的关键点。

前置检查 ​

  • 已完成 18 · 审计与形式化 与 09 · 部署与验证。
  • 理解 onlyOwner、自定义错误与事件。
  • 知道升级代理与 timelock 的基本关系。

本章示例位于 examples/19-enterprise，包含 3 个合约与 13 条测试。

1. 为什么不是“一个 owner 走天下” ​

单 owner 的风险：

  • 私钥泄露即全部资产失守。
  • 单点误操作无法回滚。
  • 升级与参数变更没有缓冲期。
  • 出事时没有明确的责任链与暂停机制。

企业级方案通常分层：

text
多签（决策）
   ↓ 提案/确认/执行
Timelock（延迟与缓冲）
   ↓ 调用
金库 / 业务合约（资产与状态）
   ↑
Guardian（只负责紧急暂停）

2. 多签：提案、确认、执行 ​

solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

/// @title MiniSafe
/// @notice 2-of-3 多签：提案、确认、达到阈值后执行。
// #region safe-core
contract MiniSafe {
    struct Transaction {
        address to;
        uint256 value;
        bytes data;
        uint256 confirmations;
        bool executed;
    }

    mapping(address => bool) public owners;
    mapping(uint256 => Transaction) public transactions;
    mapping(uint256 => mapping(address => bool)) public approved;

    uint256 public immutable threshold;
    uint256 public nonce;

    event Proposed(uint256 indexed id, address indexed proposer, address to, uint256 value);
    event Approved(uint256 indexed id, address indexed owner, uint256 confirmations);
    event Executed(uint256 indexed id, address indexed executor);

    error NotOwner();
    error AlreadyApproved();
    error NotEnoughConfirmations();
    error AlreadyExecuted();
    error CallFailed();
    error ZeroAddress();

    modifier onlyOwner() {
        if (!owners[msg.sender]) revert NotOwner();
        _;
    }

    constructor(address[3] memory initialOwners, uint256 threshold_) {
        if (threshold_ == 0 || threshold_ > 3) revert NotEnoughConfirmations();
        for (uint256 i = 0; i < 3; i++) {
            if (initialOwners[i] == address(0)) revert ZeroAddress();
            owners[initialOwners[i]] = true;
        }
        threshold = threshold_;
    }

    function propose(
        address to,
        uint256 value,
        bytes calldata data
    ) external onlyOwner returns (uint256 id) {
        if (to == address(0)) revert ZeroAddress();

        id = nonce++;
        transactions[id] = Transaction(to, value, data, 0, false);
        emit Proposed(id, msg.sender, to, value);
    }
    // #endregion

    // #region safe-execute
    function approve(uint256 id) external onlyOwner {
        Transaction storage txn = transactions[id];
        if (txn.executed) revert AlreadyExecuted();
        if (approved[id][msg.sender]) revert AlreadyApproved();

        approved[id][msg.sender] = true;
        txn.confirmations += 1;
        emit Approved(id, msg.sender, txn.confirmations);
    }

    function execute(uint256 id) external onlyOwner {
        Transaction storage txn = transactions[id];
        if (txn.executed) revert AlreadyExecuted();
        if (txn.confirmations < threshold) revert NotEnoughConfirmations();

        // 先标记已执行，再做外部调用
        txn.executed = true;

        (bool ok, ) = txn.to.call{ value: txn.value }(txn.data);
        if (!ok) revert CallFailed();

        emit Executed(id, msg.sender);
    }
    // #endregion

    receive() external payable {}
}

三个安全点：

  1. 每个提案独立 ID：nonce 递增，避免不同交易混用确认。
  2. 同一 owner 只能确认一次：approved[id][msg.sender] 防重复计数。
  3. 执行前先标记 executed：外部调用可能重入，不能等调用完成再改状态。

3. Timelock：给治理留缓冲期 ​

solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

/// @title Timelock
/// @notice 延迟执行：排队后至少等待 delay 秒，管理员可取消。
// #region timelock
contract Timelock {
    address public admin;
    uint256 public immutable delay;

    mapping(bytes32 => bool) public queued;
    mapping(bytes32 => uint256) public etaOf;

    event Queued(bytes32 indexed id, address target, uint256 value, uint256 eta);
    event Executed(bytes32 indexed id, address target, uint256 value);
    event Cancelled(bytes32 indexed id);

    error NotAdmin();
    error DelayTooShort();
    error NotQueued();
    error TooEarly();
    error CallFailed();

    modifier onlyAdmin() {
        if (msg.sender != admin) revert NotAdmin();
        _;
    }

    constructor(address admin_, uint256 delay_) {
        admin = admin_;
        delay = delay_;
    }

    function transactionId(
        address target,
        uint256 value,
        bytes calldata data,
        uint256 eta
    ) public pure returns (bytes32) {
        return keccak256(abi.encode(target, value, keccak256(data), eta));
    }
    // #endregion

    // #region timelock-queue
    function queue(
        address target,
        uint256 value,
        bytes calldata data,
        uint256 eta
    ) external onlyAdmin returns (bytes32 id) {
        if (eta < block.timestamp + delay) revert DelayTooShort();

        id = transactionId(target, value, data, eta);
        queued[id] = true;
        etaOf[id] = eta;

        emit Queued(id, target, value, eta);
    }
    // #endregion

    // #region timelock-execute
    function execute(
        address target,
        uint256 value,
        bytes calldata data,
        uint256 eta
    ) external {
        bytes32 id = transactionId(target, value, data, eta);
        if (!queued[id]) revert NotQueued();
        if (block.timestamp < eta) revert TooEarly();

        // 先消费队列项，避免重入重复执行
        queued[id] = false;

        (bool ok, ) = target.call{ value: value }(data);
        if (!ok) revert CallFailed();

        emit Executed(id, target, value);
    }

    function cancel(
        address target,
        uint256 value,
        bytes calldata data,
        uint256 eta
    ) external onlyAdmin {
        bytes32 id = transactionId(target, value, data, eta);
        if (!queued[id]) revert NotQueued();

        queued[id] = false;
        emit Cancelled(id);
    }
    // #endregion

    receive() external payable {}
}

关键参数：

  • eta：最早可执行时间，必须大于 block.timestamp + delay。
  • delay：链上不可变的缓冲期，给用户与监控系统反应时间。
  • cancel：管理员可在执行前撤销，撤掉恶意或错误提案。

eta 要留余量

测试或脚本里 eta = now + delay 往往因为交易上链消耗区块时间而变成“延迟不足”。实际部署要留出区块时间余量。

4. 金库：owner 与 guardian 分权 ​

solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

/// @title GovernedTreasury
/// @notice 由多签或 timelock 拥有的金库，guardian 可紧急暂停。
// #region treasury
contract GovernedTreasury {
    address public owner;
    address public guardian;
    bool public paused;

    event Released(address indexed to, uint256 amount);
    event Paused(address indexed by);
    event Unpaused(address indexed by);

    error NotOwner();
    error NotGuardian();
    error IsPaused();
    error CallFailed();

    modifier onlyOwner() {
        if (msg.sender != owner) revert NotOwner();
        _;
    }

    constructor(address owner_, address guardian_) payable {
        owner = owner_;
        guardian = guardian_;
    }

    function release(address payable to, uint256 amount) external onlyOwner {
        if (paused) revert IsPaused();

        (bool ok, ) = to.call{ value: amount }('');
        if (!ok) revert CallFailed();

        emit Released(to, amount);
    }

    function pause() external {
        if (msg.sender != guardian && msg.sender != owner) revert NotGuardian();
        paused = true;
        emit Paused(msg.sender);
    }

    function unpause() external onlyOwner {
        paused = false;
        emit Unpaused(msg.sender);
    }

    receive() external payable {}
}
// #endregion

非对称权限：

  • owner（多签 / timelock）：可以转账、解除暂停。
  • guardian：只能暂停，不能转账、不能解除。

这样即使 guardian 被攻破，攻击者也无法转走资金；最坏情况是短期冻结，随后由 owner 恢复。

5. 监控：事件就是运维接口 ​

建议至少埋这些事件：

事件用途
Proposed / Approved / Executed多签流程追踪
Queued / Cancelled / Executedtimelock 队列与取消
Released金库资金流出
Paused / Unpaused紧急状态
Upgraded / AdminChanged代理与权限变更

监控侧要做：

  • 大额 Released 实时告警。
  • timelock 新增队列立即通知，不等到执行。
  • owner / guardian 变更走二次确认。
  • 失败交易与反复回滚的地址加入观察名单。
  • 指标：暂停时长、队列积压、确认延迟、异常调用频率。

6. 升级治理 ​

代理 + timelock 的常见组合：

  1. 多签提出升级提案。
  2. timelock 排队，等待缓冲期。
  3. 监控与社区审查字节码、存储布局、权限变更。
  4. 到期后执行升级。
  5. 升级后做冒烟测试与回滚预案。

要点：

  • 实现合约地址变更必须发事件。
  • 存储布局必须做升级兼容检查。
  • timelock 的 admin 不能是单一 EOA。
  • 紧急升级可以缩短延迟，但不能绕过暂停与多签。

7. 密钥与运维 ​

  • owner 私钥使用硬件钱包或多方计算（MPC）。
  • 提案内容在链下先做可读化审查，避免盲签。
  • 定期演练：私钥丢失、guardian 被攻破、timelock 被塞入恶意提案。
  • 值班表与升级窗口写进 runbook，不依赖个人记忆。

8. 测试覆盖 ​

13 条测试：

  • 2/3 阈值执行、阈值不足、重复确认、非 owner 提案、重放。
  • timelock 延迟不足、未到期、到期执行、取消、非 admin。
  • 金库暂停、非 owner 直转、解除暂停后恢复。

9. 常见坑 ​

  • 多签阈值写死且无法调整：团队变化后无法轮换 owner。
  • timelock admin 是 EOA：绕过整个延迟机制。
  • guardian 能解除暂停：紧急权限变成永久接管。
  • 只监控转账，不监控队列：恶意提案在执行前无人发现。
  • 升级不做存储布局检查：新实现覆盖旧存储。
  • 事件缺失：链下系统无法重建状态。
  • 没有演练：真实事故时流程跑不通。

10. 练习题 ​

  1. 给 MiniSafe 增加 owner 轮换，并要求多签确认。
  2. 给 timelock 增加“最小延迟”与“最大延迟”双边界。
  3. 设计 guardian 多签（2-of-3），避免单点暂停。
  4. 写一个事件索引器，实时列出待执行的 timelock 提案。
  5. 写一份升级 runbook：提案、审查、执行、回滚、通知。
