1. AMM 的核心公式 ​

恒定乘积做市要求：

text
x * y = k
1

用户用 amountIn 换出 amountOut，扣 0.3% 手续费后满足：

text
(reserveIn + amountIn * 0.997) * (reserveOut - amountOut) = k'
amountOut = reserveOut * (amountIn * 0.997) / (reserveIn + amountIn * 0.997)
1
2

合约里的实现把比例写成整数：

solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import './IERC20.sol';

/// @title MiniAMM
/// @notice 恒定乘积做市：x * y = k，收取 0.3% 手续费，LP 份额按比例分配。
contract MiniAMM {
    // #region amm-liquidity
    IERC20 public immutable token0;
    IERC20 public immutable token1;

    uint256 public reserve0;
    uint256 public reserve1;

    uint256 public totalShares;
    mapping(address => uint256) public shares;

    uint256 public constant FEE_BPS = 30; // 0.3%
    uint256 public constant BPS = 10_000;

    event Mint(address indexed provider, uint256 amount0, uint256 amount1, uint256 liquidity);
    event Burn(address indexed provider, uint256 amount0, uint256 amount1, uint256 liquidity);
    event Swap(address indexed trader, bool zeroForOne, uint256 amountIn, uint256 amountOut);

    error ZeroAmount();
    error ZeroLiquidity();
    error InsufficientShares();
    error NoLiquidity();
    error ZeroOutput();
    error SlippageExceeded(uint256 minAmountOut, uint256 amountOut);

    constructor(address token0_, address token1_) {
        token0 = IERC20(token0_);
        token1 = IERC20(token1_);
    }

    function addLiquidity(uint256 amount0, uint256 amount1) external returns (uint256 liquidity) {
        if (amount0 == 0 || amount1 == 0) revert ZeroAmount();

        token0.transferFrom(msg.sender, address(this), amount0);
        token1.transferFrom(msg.sender, address(this), amount1);

        if (totalShares == 0) {
            liquidity = _sqrt(amount0 * amount1);
        } else {
            uint256 liquidity0 = (amount0 * totalShares) / reserve0;
            uint256 liquidity1 = (amount1 * totalShares) / reserve1;
            liquidity = liquidity0 < liquidity1 ? liquidity0 : liquidity1;
        }
        if (liquidity == 0) revert ZeroLiquidity();

        shares[msg.sender] += liquidity;
        totalShares += liquidity;

        _updateReserves();
        emit Mint(msg.sender, amount0, amount1, liquidity);
    }

    function removeLiquidity(uint256 liquidity) external returns (uint256 amount0, uint256 amount1) {
        if (liquidity == 0) revert ZeroAmount();
        if (shares[msg.sender] < liquidity) revert InsufficientShares();

        amount0 = (liquidity * reserve0) / totalShares;
        amount1 = (liquidity * reserve1) / totalShares;

        shares[msg.sender] -= liquidity;
        totalShares -= liquidity;

        token0.transfer(msg.sender, amount0);
        token1.transfer(msg.sender, amount1);

        _updateReserves();
        emit Burn(msg.sender, amount0, amount1, liquidity);
    }
    // #endregion

    // #region amm-swap
    function swap(uint256 amountIn, bool zeroForOne, uint256 minAmountOut)
        external
        returns (uint256 amountOut)
    {
        if (amountIn == 0) revert ZeroAmount();

        (uint256 reserveIn, uint256 reserveOut) =
            zeroForOne ? (reserve0, reserve1) : (reserve1, reserve0);
        if (reserveIn == 0 || reserveOut == 0) revert NoLiquidity();

        (IERC20 tokenIn, IERC20 tokenOut) = zeroForOne ? (token0, token1) : (token1, token0);

        tokenIn.transferFrom(msg.sender, address(this), amountIn);

        uint256 amountInWithFee = amountIn * (BPS - FEE_BPS);
        amountOut = (reserveOut * amountInWithFee) / (reserveIn * BPS + amountInWithFee);

        if (amountOut == 0) revert ZeroOutput();
        if (amountOut < minAmountOut) revert SlippageExceeded(minAmountOut, amountOut);

        tokenOut.transfer(msg.sender, amountOut);

        _updateReserves();
        emit Swap(msg.sender, zeroForOne, amountIn, amountOut);
    }
    // #endregion

    /// @notice 不变量：手续费会让 k 单调不减。
    function invariantK() external view returns (uint256) {
        return reserve0 * reserve1;
    }

    function _updateReserves() private {
        reserve0 = token0.balanceOf(address(this));
        reserve1 = token1.balanceOf(address(this));
    }

    function _sqrt(uint256 y) private pure returns (uint256 z) {
        if (y > 3) {
            z = y;
            uint256 x = y / 2 + 1;
            while (x < z) {
                z = x;
                x = (y / x + x) / 2;
            }
        } else if (y != 0) {
            z = 1;
        }
    }
}

2. 注资与 LP 份额 ​

第一次注资用几何平均：

text
liquidity = sqrt(amount0 * amount1)
1

后续注资按比例取最小值：

text
liquidity = min(amount0 * totalShares / reserve0, amount1 * totalShares / reserve1)
1

3. 不变量测试：k 不能下降 ​

手续费让 k 单调不减：

ts
let k = await amm.invariantK()
for (const amount of [1n, 5n, 20n, 3n, 50n]) {
  await amm.swap(amount * E18, zeroForOne, 0n)
  const nextK = await amm.invariantK()
  expect(nextK).to.be.greaterThanOrEqual(k)
  k = nextK
}

4. 借贷：LTV 与清算阈值 ​

MiniLending 用两个参数控制风险：

参数示例值作用
LTV70%最多能借多少
清算阈值80%何时可以被清算
清算奖励5%激励清算人

健康因子：

text
health = collateralValue * liquidationThreshold / debtValue
1
  • health >= 1：安全。
  • health < 1：可清算。
solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import './IERC20.sol';
import './MockOracle.sol';

/// @title MiniLending
/// @notice 单抵押、单债务的简化借贷：70% LTV、80% 清算阈值、5% 清算奖励。
contract MiniLending {
    // #region lending-core
    IERC20 public immutable collateral;
    IERC20 public immutable debtToken;
    MockOracle public immutable oracle;

    uint256 public constant BPS = 10_000;
    uint256 public constant LTV_BPS = 7_000;
    uint256 public constant LIQUIDATION_THRESHOLD_BPS = 8_000;
    uint256 public constant LIQUIDATION_BONUS_BPS = 500;
    uint256 public constant PRICE_SCALE = 1e18;

    mapping(address => uint256) public collateralBalance;
    mapping(address => uint256) public debtBalance;

    event CollateralDeposited(address indexed user, uint256 amount);
    event CollateralWithdrawn(address indexed user, uint256 amount);
    event Borrowed(address indexed user, uint256 amount);
    event Repaid(address indexed user, uint256 amount);
    event Liquidated(
        address indexed user,
        address indexed liquidator,
        uint256 repaid,
        uint256 seized
    );

    error ZeroAmount();
    error ExceedsLtv(uint256 limit, uint256 requested);
    error InsufficientCollateral();
    error InsufficientDebt(uint256 debt, uint256 requested);
    error HealthyPosition(uint256 healthFactor);

    constructor(address collateral_, address debtToken_, address oracle_) {
        collateral = IERC20(collateral_);
        debtToken = IERC20(debtToken_);
        oracle = MockOracle(oracle_);
    }

    function depositCollateral(uint256 amount) external {
        if (amount == 0) revert ZeroAmount();
        collateral.transferFrom(msg.sender, address(this), amount);
        collateralBalance[msg.sender] += amount;
        emit CollateralDeposited(msg.sender, amount);
    }

    function withdrawCollateral(uint256 amount) external {
        if (amount == 0) revert ZeroAmount();
        if (collateralBalance[msg.sender] < amount) revert InsufficientCollateral();

        collateralBalance[msg.sender] -= amount;

        uint256 health = healthFactor(msg.sender);
        if (health < PRICE_SCALE) revert HealthyPosition(health);

        collateral.transfer(msg.sender, amount);
        emit CollateralWithdrawn(msg.sender, amount);
    }

    function borrow(uint256 amount) external {
        if (amount == 0) revert ZeroAmount();

        uint256 limit = (collateralValue(msg.sender) * LTV_BPS) / BPS;
        uint256 requested = debtBalance[msg.sender] + amount;
        if (requested > limit) revert ExceedsLtv(limit, requested);

        debtBalance[msg.sender] = requested;
        debtToken.transfer(msg.sender, amount);
        emit Borrowed(msg.sender, amount);
    }

    function repay(uint256 amount) external {
        if (amount == 0) revert ZeroAmount();
        if (debtBalance[msg.sender] < amount) revert InsufficientDebt(debtBalance[msg.sender], amount);

        debtBalance[msg.sender] -= amount;
        debtToken.transferFrom(msg.sender, address(this), amount);
        emit Repaid(msg.sender, amount);
    }
    // #endregion

    // #region liquidation
    function liquidate(address user, uint256 repayAmount) external {
        uint256 health = healthFactor(user);
        if (health >= PRICE_SCALE) revert HealthyPosition(health);
        if (repayAmount == 0) revert ZeroAmount();
        if (debtBalance[user] < repayAmount) revert InsufficientDebt(debtBalance[user], repayAmount);

        uint256 seized =
            (repayAmount * (BPS + LIQUIDATION_BONUS_BPS) * PRICE_SCALE) /
            (BPS * oracle.price());
        if (collateralBalance[user] < seized) revert InsufficientCollateral();

        debtBalance[user] -= repayAmount;
        collateralBalance[user] -= seized;

        debtToken.transferFrom(msg.sender, address(this), repayAmount);
        collateral.transfer(msg.sender, seized);

        emit Liquidated(user, msg.sender, repayAmount, seized);
    }
    // #endregion

    function collateralValue(address user) public view returns (uint256) {
        return (collateralBalance[user] * oracle.price()) / PRICE_SCALE;
    }

    function healthFactor(address user) public view returns (uint256) {
        if (debtBalance[user] == 0) return type(uint256).max;
        return
            (collateralValue(user) * LIQUIDATION_THRESHOLD_BPS * PRICE_SCALE) /
            (BPS * debtBalance[user]);
    }
}

        collateralBalance[user] -= seized;

        debtToken.transferFrom(msg.sender, address(this), repayAmount);
        collateral.transfer(msg.sender, seized);

        emit Liquidated(user, msg.sender, repayAmount, seized);
    }
    // #endregion

    function collateralValue(address user) public view returns (uint256) {
        return (collateralBalance[user] * oracle.price()) / PRICE_SCALE;
    }

    function healthFactor(address user) public view returns (uint256) {
        if (debtBalance[user] == 0) return type(uint256).max;
        return
            (collateralValue(user) * LIQUIDATION_THRESHOLD_BPS * PRICE_SCALE) /
            (BPS * debtBalance[user]);
    }
}

6. 预言机风险 ​

MockOracle 可以任意设价，只适合测试。真实协议要面对：

  • 现货操纵：攻击者用一笔大额 swap 推偏池子价格，再触发清算。
  • 陈旧价格：预言机长时间未更新，清算基于过时价格。
  • 多源偏差：不同来源价格差距过大时应有断路器。

防线：TWAP、多源聚合、最小更新频率、价格上下限、清算延迟。

7. 无常损失 ​

LP 在价格变动时会相比“一直持有”少赚，这部分称为无常损失（impermanent loss）。它不是合约漏洞，而是做市的經濟成本。

理解方式：

  • 价格变动后，套利者会把池子恢复到新价格比例。
  • LP 持有的两种资产比例被动调整，往往在上涨时卖早了、下跌时买早了。

9. 常见坑 ​

  • 先算输出再更新 reserves：reserves 必须来自转账后的真实余额。
  • 滑点保护设为 0：生产环境等于没有保护。
  • 把 LTV 当清算阈值：两者用途不同，抵押物价值波动时需要缓冲。
  • 清算奖励过高：会吸引过度清算，伤害借款人。
  • 忽略舍入方向：协议应在对自己不利的方向向上舍入，对用户不利的方向向下舍入。
  • 单源现货预言机：这是 DeFi 被攻击最常见的入口之一。
