14 · L2 与跨链 ​

需入门约 6 学时 前置：/advanced/09-deploy-verify验证于 2026-11-04

本章目标 ​

完成这一章后，你应该能够：

  • 说清 L1 与 L2 的职责边界，以及 Optimistic Rollup 与 ZK Rollup 的差异。
  • 描述 canonical bridge 的存款、中继、取款、领取四步，并解释每一步的等待条件。
  • 用消息哈希与“已处理”集合实现重放防护。
  • 区分“排序器确认”“L1 最终性”“挑战期结束”三种时间尺度。
  • 写出资金守恒不变量测试，并识别跨链桥的主要风险来源。

前置检查 ​

  • 已完成 09 · 部署与验证。
  • 理解 nonce、交易收据与事件。
  • 知道合约地址在不同链上可以不同，不能跨链混用。

1. L1 与 L2 不是“两条平行链” ​

L2 把执行搬到更便宜的环境，但把资产结算和最终性留在 L1：

维度L1L2
作用结算、最终性、资产托管高频执行、低 gas
确认速度慢，但最终确定快，但可能被回滚
费用高低，但提现要付 L1 成本
资产来源原生资产多数是 L1 资产的映射

因此 L2 上的“余额”本质是 L1 锁仓的凭证。桥的核心任务只有两件：锁住 L1 资产、在 L2 准确记账。

2. 两类 Rollup ​

类型安全性来源提现等待典型特征
Optimistic挑战机制 + 欺诈证明挑战期（常见数分钟到数天）成本低，提现慢
ZK密码学证明证明生成与 L1 验证提现更快，证明成本高

无论哪一类，应用层都要面对同一件事：L2 上的“已成功”不等于 L1 上的“已最终”。

3. Canonical bridge 的资金流 ​

以 ETH 为例：

text
L1: 用户 deposit → 资产锁进桥合约，产生存款消息
        ↓ 等待 L1 最终性，中继者提交消息
L2: 验证消息 → 铸造等额资产给用户
        ↓ 用户使用资产
L2: 用户 burn → 销毁资产，产生取款消息
        ↓ 等待批次可用 + 挑战期结束
L1: 领取 → 释放锁仓资产

4. 消息 ID：先有身份，再谈处理 ​

桥不能靠“用户地址 + 金额”判断消息，因为它们可能重复。正确做法是给每条消息一个稳定 ID：

javascript
import { createHash } from 'node:crypto'

export class BridgeError extends Error {
  constructor(code, message) {
    super(`${code}: ${message}`)
    this.name = 'BridgeError'
    this.code = code
  }
}

function fail(code, message) {
  throw new BridgeError(code, message)
}

// #region bridge-message
/**
 * 把消息字段按固定顺序序列化，再由 sha256 生成消息 ID。
 * 字段顺序必须稳定，否则同一笔存款会得到两个不同的 ID。
 */
function canonical(payload) {
  return JSON.stringify({
    kind: payload.kind,
    nonce: String(payload.nonce),
    from: payload.from.toLowerCase(),
    to: payload.to.toLowerCase(),
    amount: String(payload.amount)
  })
}

export function messageId(payload) {
  return '0x' + createHash('sha256').update(canonical(payload)).digest('hex')
}
// #endregion

function assertMessage(message, kind) {
  if (message?.kind !== kind) {
    fail('WrongDirection', `expected ${kind}, got ${message?.kind}`)
  }

  const expected = messageId(message)
  if (expected !== message.id) {
    fail('InvalidMessage', 'message id does not match its payload')
  }
}

export class Bridge {
  #l1Locked = 0n
  #l2Supply = 0n
  #pendingDeposits = new Map()
  #pendingWithdrawals = new Map()
  #processed = new Set()

  constructor({ l1Finality = 12, challengePeriod = 7 } = {}) {
    this.l1Finality = l1Finality
    this.challengePeriod = challengePeriod
  }

  get l1Locked() {
    return this.#l1Locked
  }

  get l2Supply() {
    return this.#l2Supply
  }

  get pendingDepositAmount() {
    return [...this.#pendingDeposits.values()].reduce((sum, amount) => sum + amount, 0n)
  }

  get pendingWithdrawalAmount() {
    return [...this.#pendingWithdrawals.values()].reduce((sum, amount) => sum + amount, 0n)
  }

  // #region bridge-invariant
  /**
   * 守恒式：L1 锁仓 = L2 流通 + 在途存款 + 在途取款。
   * 任何一步状态迁移都必须让这个等式保持成立。
   */
  invariantHolds() {
    return (
      this.#l1Locked ===
      this.#l2Supply + this.pendingDepositAmount + this.pendingWithdrawalAmount
    )
  }
  // #endregion

  // #region bridge-deposit
  /**
   * L1 侧：用户把资产锁进桥合约，生成一条待中继的存款消息。
   */
  deposit({ from, to = from, amount, nonce, l1Block }) {
    const value = BigInt(amount)
    if (value <= 0n) fail('ZeroAmount', 'deposit amount must be positive')
    if (typeof from !== 'string' || typeof to !== 'string') {
      fail('BadAddress', 'from and to are required')
    }

    const message = {
      kind: 'deposit',
      from,
      to,
      amount: value,
      nonce: String(nonce),
      l1Block: Number(l1Block)
    }
    message.id = messageId(message)

    if (this.#pendingDeposits.has(message.id) || this.#processed.has(message.id)) {
      fail('DuplicateMessage', `message ${message.id} already exists`)
    }

    this.#pendingDeposits.set(message.id, value)
    this.#l1Locked += value
    return message
  }
  // #endregion

  // #region bridge-relay
  /**
   * L2 侧：中继者只有在 L1 达到最终性之后才能铸币。
   * 已处理过的消息会被拒绝，防止同一笔存款被重复铸造。
   */
  relayDeposit(message, { l1Head }) {
    assertMessage(message, 'deposit')

    if (this.#processed.has(message.id)) {
      fail('ReplayDetected', `deposit ${message.id} was already relayed`)
    }

    if (Number(l1Head) < message.l1Block + this.l1Finality) {
      fail('NotFinal', `deposit needs ${this.l1Finality} L1 confirmations`)
    }

    const pending = this.#pendingDeposits.get(message.id)
    if (pending === undefined) {
      fail('UnknownMessage', `deposit ${message.id} is not pending`)
    }
    if (pending !== message.amount) {
      fail('InvalidMessage', 'amount does not match the pending deposit')
    }

    this.#pendingDeposits.delete(message.id)
    this.#processed.add(message.id)
    this.#l2Supply += message.amount
    return { id: message.id, minted: message.amount, l2Supply: this.#l2Supply }
  }
  // #endregion

  // #region bridge-burn
  /**
   * L2 侧：销毁代表资产并登记一条取款消息，起算挑战期。
   */
  burn({ from, amount, nonce, l2Block, l1Block }) {
    const value = BigInt(amount)
    if (value <= 0n) fail('ZeroAmount', 'burn amount must be positive')
    if (value > this.#l2Supply) fail('InsufficientSupply', 'burn exceeds L2 supply')

    const withdrawal = {
      kind: 'withdrawal',
      from,
      to: from,
      amount: value,
      nonce: String(nonce),
      l2Block: Number(l2Block),
      l1Block: Number(l1Block),
      availableAtL1Block: Number(l1Block) + this.challengePeriod
    }
    withdrawal.id = messageId(withdrawal)

    if (this.#pendingWithdrawals.has(withdrawal.id) || this.#processed.has(withdrawal.id)) {
      fail('DuplicateMessage', `message ${withdrawal.id} already exists`)
    }

    this.#pendingWithdrawals.set(withdrawal.id, value)
    this.#l2Supply -= value
    return withdrawal
  }
  // #endregion

  // #region bridge-claim
  /**
   * L1 侧：挑战期结束且消息未被处理过，才释放锁仓资产。
   */
  claimWithdrawal(withdrawal, { l1Head }) {
    assertMessage(withdrawal, 'withdrawal')

    if (this.#processed.has(withdrawal.id)) {
      fail('ReplayDetected', `withdrawal ${withdrawal.id} was already claimed`)
    }

    if (Number(l1Head) < withdrawal.availableAtL1Block) {
      fail('NotFinal', `withdrawal is claimable at L1 block ${withdrawal.availableAtL1Block}`)
    }

    const pending = this.#pendingWithdrawals.get(withdrawal.id)
    if (pending === undefined) {
      fail('UnknownMessage', `withdrawal ${withdrawal.id} is not pending`)
    }
    if (pending !== withdrawal.amount) {
      fail('InvalidMessage', 'amount does not match the pending withdrawal')
    }

    this.#pendingWithdrawals.delete(withdrawal.id)
    this.#processed.add(withdrawal.id)
    this.#l1Locked -= withdrawal.amount
    return { id: withdrawal.id, released: withdrawal.amount, l1Locked: this.#l1Locked }
  }
  // #endregion
}

要点：

  • 字段顺序固定，地址统一小写，金额统一字符串。
  • 同一个消息在 L1 和 L2 计算出的 ID 必须一致。
  • 校验时重新计算 ID，字段被改动就会暴露。

ID 不是签名

消息 ID 只能防止“字段被悄悄改动”，不能证明消息由合法桥发出。生产环境还需要签名、Merkle 证明或状态根验证。

deposit 做了三件事：

  1. 校验金额与地址，拒绝零金额。
  2. 生成消息并登记为“在途存款”。
  3. 增加 L1 锁仓量。

注意此时 L2 还没有铸币，所以在途金额必须单独记账，否则守恒式会立刻失真。

6. 中继：等最终性，且只处理一次 ​

三个检查顺序很重要：

  1. 方向：存款消息只能走存款通道。
  2. 重放：已处理的消息直接拒绝，不能等检查完余额再说。
  3. 最终性：l1Head 必须达到 l1Block + l1Finality。

如果先检查余额再检查重放，攻击者可以构造“余额刚好够”的重复消息，让错误信息掩盖真正的问题。安全代码要把“不可逆拒绝”放在前面。

8. 守恒不变量 ​

桥最值得写成测试的性质不是“某次转账成功”，而是资金守恒：

javascript
import { createHash } from 'node:crypto'

export class BridgeError extends Error {
  constructor(code, message) {
    super(`${code}: ${message}`)
    this.name = 'BridgeError'
    this.code = code
  }
}

function fail(code, message) {
  throw new BridgeError(code, message)
}

// #region bridge-message
/**
 * 把消息字段按固定顺序序列化，再由 sha256 生成消息 ID。
 * 字段顺序必须稳定，否则同一笔存款会得到两个不同的 ID。
 */
function canonical(payload) {
  return JSON.stringify({
    kind: payload.kind,
    nonce: String(payload.nonce),
    from: payload.from.toLowerCase(),
    to: payload.to.toLowerCase(),
    amount: String(payload.amount)
  })
}

export function messageId(payload) {
  return '0x' + createHash('sha256').update(canonical(payload)).digest('hex')
}
// #endregion

function assertMessage(message, kind) {
  if (message?.kind !== kind) {
    fail('WrongDirection', `expected ${kind}, got ${message?.kind}`)
  }

  const expected = messageId(message)
  if (expected !== message.id) {
    fail('InvalidMessage', 'message id does not match its payload')
  }
}

export class Bridge {
  #l1Locked = 0n
  #l2Supply = 0n
  #pendingDeposits = new Map()
  #pendingWithdrawals = new Map()
  #processed = new Set()

  constructor({ l1Finality = 12, challengePeriod = 7 } = {}) {
    this.l1Finality = l1Finality
    this.challengePeriod = challengePeriod
  }

  get l1Locked() {
    return this.#l1Locked
  }

  get l2Supply() {
    return this.#l2Supply
  }

  get pendingDepositAmount() {
    return [...this.#pendingDeposits.values()].reduce((sum, amount) => sum + amount, 0n)
  }

  get pendingWithdrawalAmount() {
    return [...this.#pendingWithdrawals.values()].reduce((sum, amount) => sum + amount, 0n)
  }

  // #region bridge-invariant
  /**
   * 守恒式：L1 锁仓 = L2 流通 + 在途存款 + 在途取款。
   * 任何一步状态迁移都必须让这个等式保持成立。
   */
  invariantHolds() {
    return (
      this.#l1Locked ===
      this.#l2Supply + this.pendingDepositAmount + this.pendingWithdrawalAmount
    )
  }
  // #endregion

  // #region bridge-deposit
  /**
   * L1 侧：用户把资产锁进桥合约，生成一条待中继的存款消息。
   */
  deposit({ from, to = from, amount, nonce, l1Block }) {
    const value = BigInt(amount)
    if (value <= 0n) fail('ZeroAmount', 'deposit amount must be positive')
    if (typeof from !== 'string' || typeof to !== 'string') {
      fail('BadAddress', 'from and to are required')
    }

    const message = {
      kind: 'deposit',
      from,
      to,
      amount: value,
      nonce: String(nonce),
      l1Block: Number(l1Block)
    }
    message.id = messageId(message)

    if (this.#pendingDeposits.has(message.id) || this.#processed.has(message.id)) {
      fail('DuplicateMessage', `message ${message.id} already exists`)
    }

    this.#pendingDeposits.set(message.id, value)
    this.#l1Locked += value
    return message
  }
  // #endregion

  // #region bridge-relay
  /**
   * L2 侧：中继者只有在 L1 达到最终性之后才能铸币。
   * 已处理过的消息会被拒绝，防止同一笔存款被重复铸造。
   */
  relayDeposit(message, { l1Head }) {
    assertMessage(message, 'deposit')

    if (this.#processed.has(message.id)) {
      fail('ReplayDetected', `deposit ${message.id} was already relayed`)
    }

    if (Number(l1Head) < message.l1Block + this.l1Finality) {
      fail('NotFinal', `deposit needs ${this.l1Finality} L1 confirmations`)
    }

    const pending = this.#pendingDeposits.get(message.id)
    if (pending === undefined) {
      fail('UnknownMessage', `deposit ${message.id} is not pending`)
    }
    if (pending !== message.amount) {
      fail('InvalidMessage', 'amount does not match the pending deposit')
    }

    this.#pendingDeposits.delete(message.id)
    this.#processed.add(message.id)
    this.#l2Supply += message.amount
    return { id: message.id, minted: message.amount, l2Supply: this.#l2Supply }
  }
  // #endregion

  // #region bridge-burn
  /**
   * L2 侧：销毁代表资产并登记一条取款消息，起算挑战期。
   */
  burn({ from, amount, nonce, l2Block, l1Block }) {
    const value = BigInt(amount)
    if (value <= 0n) fail('ZeroAmount', 'burn amount must be positive')
    if (value > this.#l2Supply) fail('InsufficientSupply', 'burn exceeds L2 supply')

    const withdrawal = {
      kind: 'withdrawal',
      from,
      to: from,
      amount: value,
      nonce: String(nonce),
      l2Block: Number(l2Block),
      l1Block: Number(l1Block),
      availableAtL1Block: Number(l1Block) + this.challengePeriod
    }
    withdrawal.id = messageId(withdrawal)

    if (this.#pendingWithdrawals.has(withdrawal.id) || this.#processed.has(withdrawal.id)) {
      fail('DuplicateMessage', `message ${withdrawal.id} already exists`)
    }

    this.#pendingWithdrawals.set(withdrawal.id, value)
    this.#l2Supply -= value
    return withdrawal
  }
  // #endregion

  // #region bridge-claim
  /**
   * L1 侧：挑战期结束且消息未被处理过，才释放锁仓资产。
   */
  claimWithdrawal(withdrawal, { l1Head }) {
    assertMessage(withdrawal, 'withdrawal')

    if (this.#processed.has(withdrawal.id)) {
      fail('ReplayDetected', `withdrawal ${withdrawal.id} was already claimed`)
    }

    if (Number(l1Head) < withdrawal.availableAtL1Block) {
      fail('NotFinal', `withdrawal is claimable at L1 block ${withdrawal.availableAtL1Block}`)
    }

    const pending = this.#pendingWithdrawals.get(withdrawal.id)
    if (pending === undefined) {
      fail('UnknownMessage', `withdrawal ${withdrawal.id} is not pending`)
    }
    if (pending !== withdrawal.amount) {
      fail('InvalidMessage', 'amount does not match the pending withdrawal')
    }

    this.#pendingWithdrawals.delete(withdrawal.id)
    this.#processed.add(withdrawal.id)
    this.#l1Locked -= withdrawal.amount
    return { id: withdrawal.id, released: withdrawal.amount, l1Locked: this.#l1Locked }
  }
  // #endregion
}

L1 锁仓 = L2 流通 + 在途存款 + 在途取款

四个动作对等式的影响：

动作L1 锁仓L2 流通在途存款在途取款
deposit+amount不变+amount不变
relay不变+amount-amount不变
burn不变-amount不变+amount
claim-amount不变不变-amount

任何一行写错，等式都会在某一步破裂。示例的随机序列测试就是不断触发这四类迁移并逐步断言。

9. L2 gas 与部署差异 ​

事项L1L2
gas 价格高，波动大低，但 L1 提交成本会分摊
部署一次交易通常同样是一次交易，但需确认浏览器支持
验证源码Etherscan 等各 L2 的浏览器与验证 API 不同
地址各链独立同一私钥在不同链上地址相同，但合约地址取决于部署者与 nonce

实践建议：

  • 部署脚本按 chainId 分文件维护地址簿，不要复用同一个地址常量。
  • 前端同时展示 L1 与 L2 的确认状态，不要让用户误以为“L2 成功就等于最终”。
  • 跨链转账要展示预计等待时间与失败回退路径。

10. 跨链风险清单 ​

  • 消息重放：同一签名或同一证明被用两次。
  • 最终性误判：L2 已确认但 L1 批次被回滚。
  • 验证缺失：只校验格式，不校验来源。
  • 桥合约权限过大：升级或提款权限没有 timelock / 多签。
  • 流动性错配：锁仓量与 L2 供应量长期偏离。
  • 预言机与跨链组合：价格源和消息源同时被操纵。
  • 用户界面误导：把“已提交”显示成“已到账”。

12. 常见坑 ​

  • 只按地址和金额去重：同一用户可能连续转相同金额，必须用消息 ID + nonce。
  • 先铸币再等最终性：L1 回滚后 L2 已经花掉了资产，桥出现坏账。
  • 挑战期用 L2 区块计算：挑战期属于 L1 时间线，混用时钟会提前放款。
  • 把在途资金漏出等式：存款已锁但未铸、取款已烧但未放，都是真实负债。
  • 忽略失败路径：中继失败、领取失败、批次缺失都要有重试与告警。
