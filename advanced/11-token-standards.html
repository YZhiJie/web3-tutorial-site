11 · Token 标准 ​

需入门约 6 学时 前置：/advanced/07-solidity-patterns验证于 2026-11-04

本章目标 ​

完成这一章后，你应该能够：

  • 说清 ERC-20、ERC-721、ERC-1155 的适用场景与核心接口差异。
  • 实现一个可读的最小 ERC-721，包括元数据、单 token 授权与 operator 授权。
  • 解释为什么转移前要清空单 token 授权。
  • 写一个托管型固定价市场，处理上架、成交、抽费、取消与重复上架。
  • 识别授权与托管模式各自的信任边界。

前置检查 ​

  • 已完成 07 · Solidity 进阶与合约模式。
  • 理解 msg.sender、事件与自定义错误。
  • 能读懂 ERC-20 的 approve / transferFrom。

本章示例位于 examples/11-token-standards，2 个合约、14 条测试。

1. 三种标准怎么选 ​

标准表示典型用途批量转移
ERC-20同质化余额积分、稳定币、治理币无原生批量
ERC-721唯一 tokenId收藏品、身份、门票逐个转移
ERC-1155id + 数量游戏道具、多版本资产原生批量

本章只实现 ERC-721。ERC-20 已在第 05 章用 MiniToken 演示过；ERC-1155 的核心是“每种 id 一个余额映射”，可以作为练习。

2. ERC-721 的核心状态 ​

solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

/// @title MiniNFT
/// @notice 教学用最小 ERC-721：元数据、单 token 授权、全局 operator 与转移。
/// @dev 未实现 ERC-165 与 safeTransferFrom，生产项目请使用审计过的库。
contract MiniNFT {
    // #region erc721-core
    string public name;
    string public symbol;
    string public baseURI;

    address public immutable owner;

    uint256 private _nextTokenId = 1;
    mapping(uint256 => address) private _owners;
    mapping(address => uint256) private _balances;

    event Transfer(address indexed from, address indexed to, uint256 indexed tokenId);

    error NotOwner(address caller);
    error ZeroAddress();
    error TokenNotExists(uint256 tokenId);

    constructor(string memory name_, string memory symbol_, string memory baseURI_) {
        name = name_;
        symbol = symbol_;
        baseURI = baseURI_;
        owner = msg.sender;
    }

    modifier onlyOwner() {
        if (msg.sender != owner) revert NotOwner(msg.sender);
        _;
    }

    function mint(address to) external onlyOwner returns (uint256 tokenId) {
        if (to == address(0)) revert ZeroAddress();

        tokenId = _nextTokenId++;
        _owners[tokenId] = to;
        _balances[to] += 1;

        emit Transfer(address(0), to, tokenId);
    }

    function ownerOf(uint256 tokenId) public view returns (address) {
        address tokenOwner = _owners[tokenId];
        if (tokenOwner == address(0)) revert TokenNotExists(tokenId);
        return tokenOwner;
    }

    function balanceOf(address account) external view returns (uint256) {
        return _balances[account];
    }

    function tokenURI(uint256 tokenId) external view returns (string memory) {
        ownerOf(tokenId);
        return string(abi.encodePacked(baseURI, _toString(tokenId), '.json'));
    }

    function setBaseURI(string calldata newBaseURI) external onlyOwner {
        baseURI = newBaseURI;
    }
    // #endregion

    // #region erc721-approve
    mapping(uint256 => address) private _tokenApprovals;
    mapping(address => mapping(address => bool)) private _operatorApprovals;

    event Approval(address indexed owner, address indexed approved, uint256 indexed tokenId);
    event ApprovalForAll(address indexed owner, address indexed operator, bool approved);

    function approve(address to, uint256 tokenId) external {
        address tokenOwner = ownerOf(tokenId);
        require(msg.sender == tokenOwner || isApprovedForAll(tokenOwner, msg.sender), 'not authorized');

        _tokenApprovals[tokenId] = to;
        emit Approval(tokenOwner, to, tokenId);
    }

    function getApproved(uint256 tokenId) public view returns (address) {
        ownerOf(tokenId);
        return _tokenApprovals[tokenId];
    }

    function setApprovalForAll(address operator, bool approved) external {
        _operatorApprovals[msg.sender][operator] = approved;
        emit ApprovalForAll(msg.sender, operator, approved);
    }

    function isApprovedForAll(address tokenOwner, address operator) public view returns (bool) {
        return _operatorApprovals[tokenOwner][operator];
    }
    // #endregion

    // #region erc721-transfer
    function transferFrom(address from, address to, uint256 tokenId) public {
        address tokenOwner = ownerOf(tokenId);
        require(tokenOwner == from, 'wrong from');
        require(to != address(0), 'zero address');
        require(
            msg.sender == tokenOwner ||
                _tokenApprovals[tokenId] == msg.sender ||
                isApprovedForAll(tokenOwner, msg.sender),
            'not authorized'
        );

        // 转移前清空单 token 授权，避免新 owner 的资产被旧授权转走
        _tokenApprovals[tokenId] = address(0);

        _balances[from] -= 1;
        _balances[to] += 1;
        _owners[tokenId] = to;

        emit Transfer(from, to, tokenId);
    }
    // #endregion

    function _toString(uint256 value) private pure returns (string memory) {
        if (value == 0) return '0';

        uint256 temp = value;
        uint256 digits;
        while (temp != 0) {
            digits++;
            temp /= 10;
        }

        bytes memory buffer = new bytes(digits);
        while (value != 0) {
            digits -= 1;
            buffer[digits] = bytes1(uint8(48 + (value % 10)));
            value /= 10;
        }
        return string(buffer);
    }
}

四个映射对应四个问题：

  • _owners[tokenId]：这个 token 现在归谁？
  • _balances[account]：这个账户有几个 token？
  • _tokenApprovals[tokenId]：谁被允许转走这一个 token？
  • _operatorApprovals[owner][operator]：谁被允许管理这个账户的全部 token？

tokenURI 只是元数据入口，链上不存图片。示例用 baseURI + tokenId + ".json" 拼接；生产项目通常把 JSON 放到 IPFS，并在 JSON 里引用图片。

无限授权的边界

setApprovalForAll 一旦给错地址，对方可以转走该账户全部 NFT。前端应明确展示授权范围，并提供撤销入口。

interface IERC721 {
    function ownerOf(uint256 tokenId) external view returns (address);

    function getApproved(uint256 tokenId) external view returns (address);

    function isApprovedForAll(address owner, address operator) external view returns (bool);

    function transferFrom(address from, address to, uint256 tokenId) external;
}
// #endregion

上架时把 NFT 转到市场，成交时市场再转给买家：

solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import './IERC721.sol';

/// @title FixedPriceMarket
/// @notice 固定价 NFT 市场：上架时托管 NFT，成交后抽 2.5% 费用。
contract FixedPriceMarket {
    // #region market-listing
    struct Listing {
        address seller;
        uint256 price;
        bool active;
    }

    IERC721 public immutable nft;
    address public immutable feeRecipient;
    uint256 public constant FEE_BPS = 250; // 2.5%

    mapping(uint256 => Listing) public listings;

    event Listed(uint256 indexed tokenId, address indexed seller, uint256 price);
    event Sold(uint256 indexed tokenId, address indexed buyer, uint256 price, uint256 fee);
    event Cancelled(uint256 indexed tokenId, address indexed seller);

    error NotTokenOwner(address caller, uint256 tokenId);
    error MarketNotApproved(uint256 tokenId);
    error InvalidPrice();
    error AlreadyListed(uint256 tokenId);
    error NotListed(uint256 tokenId);
    error NotSeller(address caller);
    error WrongPayment(uint256 expected, uint256 actual);
    error CannotBuyOwnListing();

    function list(uint256 tokenId, uint256 price) external {
        if (listings[tokenId].active) revert AlreadyListed(tokenId);
        if (nft.ownerOf(tokenId) != msg.sender) revert NotTokenOwner(msg.sender, tokenId);
        if (price == 0) revert InvalidPrice();

        bool approved =
            nft.getApproved(tokenId) == address(this) || nft.isApprovedForAll(msg.sender, address(this));
        if (!approved) revert MarketNotApproved(tokenId);

        listings[tokenId] = Listing({seller: msg.sender, price: price, active: true});

        // 托管 NFT，成交时市场才能直接转给买家
        nft.transferFrom(msg.sender, address(this), tokenId);

        emit Listed(tokenId, msg.sender, price);
    }
    // #endregion

    // #region market-buy
    function buy(uint256 tokenId) external payable {
        Listing memory listing = listings[tokenId];
        if (!listing.active) revert NotListed(tokenId);
        if (msg.sender == listing.seller) revert CannotBuyOwnListing();
        if (msg.value != listing.price) revert WrongPayment(listing.price, msg.value);

        // 先改状态，再做外部调用
        listings[tokenId].active = false;

        uint256 fee = (listing.price * FEE_BPS) / 10_000;
        uint256 proceeds = listing.price - fee;

        nft.transferFrom(address(this), msg.sender, tokenId);

        (bool sellerOk, ) = listing.seller.call{value: proceeds}('');
        require(sellerOk, 'seller transfer failed');

        (bool feeOk, ) = feeRecipient.call{value: fee}('');
        require(feeOk, 'fee transfer failed');

        emit Sold(tokenId, msg.sender, listing.price, fee);
    }

    function cancel(uint256 tokenId) external {
        Listing memory listing = listings[tokenId];
        if (!listing.active) revert NotListed(tokenId);
        if (msg.sender != listing.seller) revert NotSeller(msg.sender);

        listings[tokenId].active = false;

        nft.transferFrom(address(this), listing.seller, tokenId);

        emit Cancelled(tokenId, listing.seller);
    }
    // #endregion

    constructor(address nft_, address feeRecipient_) {
        nft = IERC721(nft_);
        feeRecipient = feeRecipient_;
    }
}

托管模式的特点：

  • 卖家挂单后无法再把 NFT 转走，买家更有确定性。
  • 市场合约成为临时持有人，用户需要信任市场不会卷走资产。
  • 取消时由市场转回卖家。

另一种模式是“非托管挂单”：NFT 留在卖家手里，成交时依赖卖家授权，市场只做撮合。它降低托管风险，但需要处理卖家撤销授权、并发成交等问题。

6. 成交：先改状态，再转账 ​

solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import './IERC721.sol';

/// @title FixedPriceMarket
/// @notice 固定价 NFT 市场：上架时托管 NFT，成交后抽 2.5% 费用。
contract FixedPriceMarket {
    // #region market-listing
    struct Listing {
        address seller;
        uint256 price;
        bool active;
    }

    IERC721 public immutable nft;
    address public immutable feeRecipient;
    uint256 public constant FEE_BPS = 250; // 2.5%

    mapping(uint256 => Listing) public listings;

    event Listed(uint256 indexed tokenId, address indexed seller, uint256 price);
    event Sold(uint256 indexed tokenId, address indexed buyer, uint256 price, uint256 fee);
    event Cancelled(uint256 indexed tokenId, address indexed seller);

    error NotTokenOwner(address caller, uint256 tokenId);
    error MarketNotApproved(uint256 tokenId);
    error InvalidPrice();
    error AlreadyListed(uint256 tokenId);
    error NotListed(uint256 tokenId);
    error NotSeller(address caller);
    error WrongPayment(uint256 expected, uint256 actual);
    error CannotBuyOwnListing();

    function list(uint256 tokenId, uint256 price) external {
        if (listings[tokenId].active) revert AlreadyListed(tokenId);
        if (nft.ownerOf(tokenId) != msg.sender) revert NotTokenOwner(msg.sender, tokenId);
        if (price == 0) revert InvalidPrice();

        bool approved =
            nft.getApproved(tokenId) == address(this) || nft.isApprovedForAll(msg.sender, address(this));
        if (!approved) revert MarketNotApproved(tokenId);

        listings[tokenId] = Listing({seller: msg.sender, price: price, active: true});

        // 托管 NFT，成交时市场才能直接转给买家
        nft.transferFrom(msg.sender, address(this), tokenId);

        emit Listed(tokenId, msg.sender, price);
    }
    // #endregion

    // #region market-buy
    function buy(uint256 tokenId) external payable {
        Listing memory listing = listings[tokenId];
        if (!listing.active) revert NotListed(tokenId);
        if (msg.sender == listing.seller) revert CannotBuyOwnListing();
        if (msg.value != listing.price) revert WrongPayment(listing.price, msg.value);

        // 先改状态，再做外部调用
        listings[tokenId].active = false;

        uint256 fee = (listing.price * FEE_BPS) / 10_000;
        uint256 proceeds = listing.price - fee;

        nft.transferFrom(address(this), msg.sender, tokenId);

        (bool sellerOk, ) = listing.seller.call{value: proceeds}('');
        require(sellerOk, 'seller transfer failed');

        (bool feeOk, ) = feeRecipient.call{value: fee}('');
        require(feeOk, 'fee transfer failed');

        emit Sold(tokenId, msg.sender, listing.price, fee);
    }

    function cancel(uint256 tokenId) external {
        Listing memory listing = listings[tokenId];
        if (!listing.active) revert NotListed(tokenId);
        if (msg.sender != listing.seller) revert NotSeller(msg.sender);

        listings[tokenId].active = false;

        nft.transferFrom(address(this), listing.seller, tokenId);

        emit Cancelled(tokenId, listing.seller);
    }
    // #endregion

    constructor(address nft_, address feeRecipient_) {
        nft = IERC721(nft_);
        feeRecipient = feeRecipient_;
    }
}

8. 常见坑 ​

  • 转移后忘记清空授权：新 owner 的资产可能被旧授权转走。
  • 只实现 approve 不支持 operator：主流市场无法集成。
  • tokenURI 没有存在性检查：不存在的 token 也能返回 URI，误导前端。
  • 市场抽费没有上限：管理员可以改成 100%，应在部署时固定或加 timelock。
  • 用 transfer 而不是 call 付款：transfer 的固定 gas 可能让智能合约卖家无法收款。
  • 忽略 ERC-165：其他合约无法通过标准接口查询你是否支持 ERC-721。

9. 练习题 ​

  1. 给 MiniNFT 增加 burn(uint256)，只允许 owner 或 approved 调用，并清空余额与授权。
  2. 实现 safeTransferFrom，接收方是合约时调用 onERC721Received，返回错误选择器则 revert。
  3. 把市场改成非托管模式，写出“卖家在成交前撤销授权”的测试。
  4. 增加一个 1% 的版税接收方，说明它和平台费用在分账顺序上的差异。
